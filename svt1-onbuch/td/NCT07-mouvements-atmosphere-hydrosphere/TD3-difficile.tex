\begin{exercice}[Bilan radiatif, cellules de convection et circulation générale]
\label{exo:circulation_atmo}
L'atmosphère fonctionne comme une machine thermique animée par un gradient radiatif entre l'équateur et les pôles. On considère un modèle simplifié de la Terre (sphère homogène, rotation uniforme, axe perpendiculaire au plan de l'écliptique).

\begin{enumerate}
    \item \textbf{Bilan radiatif zonal.} Le flux solaire moyen absorbé par unité de surface à la latitude $\varphi$ s'écrit $F_{\text{abs}}(\varphi) = \frac{S_0}{4}(1-\alpha)\cos\varphi$, avec $S_0 = \SI{1361}{W.m^{-2}}$ (constante solaire) et $\alpha = 0,3$ (albédo planétaire moyen). Le flux infrarouge émis vers l'espace est approximé par $F_{\text{emit}}(\varphi) = A + B\cos\varphi$, avec $A = \SI{210}{W.m^{-2}}$ et $B = \SI{60}{W.m^{-2}}$.
    \begin{enumerate}
        \item Exprimez le bilan radiatif net $R(\varphi) = F_{\text{abs}}(\varphi) - F_{\text{emit}}(\varphi)$. Calculez les latitudes $\varphi_0$ où $R(\varphi_0)=0$ (frontières théoriques entre excédent et déficit énergétique). \textit{(2 pts)}
        \item En l'absence de transport méridien d'énergie, déduisez l'évolution théorique de la température de surface en fonction de la latitude. Pourquoi ce modèle est-il physiquement inacceptable pour la Terre réelle ? \textit{(1 pt)}
    \end{enumerate}

    \item \textbf{Cellules de Hadley et conservation du moment cinétique.} Dans la cellule de Hadley, l'air monte à l'équateur, diverge en altitude vers les pôles, et subit la déviation de Coriolis.
    \begin{enumerate}
        \item En supposant une conservation du moment cinétique absolu $m = \Omega a^2 \cos^2\varphi + u a \cos\varphi$ (où $\Omega$ est la vitesse de rotation terrestre, $a$ le rayon terrestre, $u$ le vent zonal relatif), montrez que le vent zonal $u$ en altitude à la latitude $\varphi$ pour une particule partie de l'équateur ($u=0, \varphi=0$) vaut $u(\varphi) = \Omega a \frac{\sin^2\varphi}{\cos\varphi}$.
        \item Calculez $u$ à $\varphi = \pm 30^\circ$. Comparez cette valeur théorique aux vents observés (courants-jets subtropicaux). Expliquez l'écart majeur par l'action des tourbillons baroclines (cellules de Ferrel). \textit{(2 pts)}
    \end{enumerate}

    \item \textbf{Interaction Océan-Atmosphère : Courants de bord ouest.} La vorticité potentielle $q = \frac{f + \zeta}{h}$ est conservée pour un fluide homogène ($f=2\Omega\sin\varphi$ paramètre de Coriolis, $\zeta$ vorticité relative, $h$ épaisseur de la couche active).
    \begin{enumerate}
        \item Expliquez pourquoi la conservation de $q$ impose l'intensification des courants de bord ouest (ex: Gulf Stream, Kuroshio) par rapport aux bord est, dans un bassin océanique soumis aux vents alizés et contre-alizés (pompage d'Ekman). \textit{(2 pts)}
        \item Quel est le rôle de ces courants dans le transport méridien de chaleur total (Atmosphère + Océan) ? Donnez un ordre de grandeur du transport maximal en PW (\SI{e15}{W}). \textit{(1 pt)}
    \end{enumerate}
\end{enumerate}
\end{exercice}

\begin{exercice}[Dynamique de l'ENSO et téléconnexions climatiques]
\label{exo:enso}
Le phénomène El Niño - Oscillation Australe (ENSO) résulte d'un couplage instable entre l'océan équatorial Pacifique et l'atmosphère (circulation de Walker).

\begin{enumerate}
    \item \textbf{Mécanisme de Bjerknes (rétroaction positive).} À partir de la figure schématique ci-après (décrite textuellement), décrivez la chaîne causale reliant une anomalie positive de température de surface (SST) dans le Pacifique centre-est à l'affaiblissement des alizés, puis au creusement de la thermocline à l'est, amplifiant l'anomalie SST initiale.
    \begin{center}
    \begin{tabular}{|c|c|c|}
    \hline
    \textbf{Phase} & \textbf{Pacifique Ouest (Indonésie)} & \textbf{Pacifique Est (Côte Amérique S)} \\
    \hline
    \textit{Normal} & Eau chaude ($>$ \SI{28}{\celsius}), convection forte, P bas & Eau froide (remontée), thermocline peu profonde, P haut \\
    \textit{El Niño} & Eau moins chaude, convection déplacée, P monte & Eau chaude, thermocline profonde, convection, P baisse \\
    \hline
    \end{tabular}
    \end{center}
    \textit{(2 pts)}

    \item \textbf{Ondes équatoriales et transition La Niña.} La fin d'un épisode El Niño est souvent déclenchée par la réflexion d'ondes de Rossby équatoriales sur la bordure ouest du bassin, qui reviennent sous forme d'ondes de Kelvin vers l'est.
    \begin{enumerate}
        \item Rapportez la relation de dispersion pour les ondes de Kelvin équatoriales ($c = \sqrt{g'H}$, $g'$ gravité réduite, $H$ épaisseur équivalente). Donnez l'ordre de grandeur de la célérité $c$ pour la première mode barocline ($H \approx \SI{200}{m}, g' \approx \SI{0.03}{m.s^{-2}}$). \textit{(1 pt)}
        \item Expliquez pourquoi ces ondes de Kelvin "froides" (abaissement de la thermocline) propagées vers l'est initient la phase froide (La Niña). \textit{(1 pt)}
    \end{enumerate}

    \item \textbf{Téléconnexion : Impact sur la mousson africaine.} Lors d'un El Niño fort, on observe souvent un déficit pluviométrique au Sahel.
    \begin{enumerate}
        \item En utilisant la circulation de Walker modifiée (ascendance déplacée vers le Pacifique centre), expliquez dynamiquement la subsidence compensatrice sur l'Atlantique tropical et l'Afrique de l'Ouest.
        \item Proposez une hypothèse sur l'évolution future de cette téléconnexion dans un contexte de réchauffement climatique (+$\SI{2}{\celsius}$ global), en considérant le gradient de température terre-mer et la contrainte de Clausius-Clapeyron. \textit{(2 pts)}
    \end{enumerate}
\end{enumerate}
\end{exercice}

\begin{exercice}[Genèse, intensification et impacts des cyclones tropicaux]
\label{exo:cyclones}
Un cyclone tropical est une machine thermique de Carnot alimentée par les flux d'enthalpie (chaleur latente + sensible) à l'interface air-mer.

\begin{enumerate}
    \item \textbf{Conditions nécessaires à la cyclogenèse.} Listez les \textbf{six} conditions dynamiques et thermodynamiques classiques (critères de Gray). Pour chacune, précisez le seuil quantitatif typique (ex: SST $>$ \SI{26.5}{\celsius} sur \SI{50}{m} de profondeur). \textit{(2 pts)}

    \item \textbf{Intensité potentielle maximale (Emanuel, 1986).} La vitesse maximale des vents $V_{\text{max}}$ s'approche de :
    $$ V_{\text{max}}^2 = \frac{C_k}{C_D} \frac{T_s - T_o}{T_o} (k^*_s - k) $$
    où $C_k, C_D$ sont les coefficients d'échange enthalpie/moment, $T_s$ température de surface (K), $T_o$ température au sommet de la troposphère (K), $k^*_s$ enthalpie de saturation à la surface, $k$ enthalpie de l'air limite.
    \begin{enumerate}
        \item Analysez la sensibilité de $V_{\text{max}}$ à une augmentation de $\SI{1}{\celsius}$ de $T_s$ (autres paramètres constants). Le lien est-il linéaire ? \textit{(1 pt)}
        \item Discutez l'effet paradoxal d'un réchauffement climatique uniforme ($T_s$ et $T_o$ augmentent de même amplitude) sur $V_{\text{max}}$. \textit{(1 pt)}
    \end{enumerate}

    \item \textbf{Trajectoire et risque côtière (Cas d'étude : Golfe de Guinée).} Bien que rares, des systèmes dépressionnaires tropicaux peuvent se former près du Golfe de Guinée (ex: tempête tropicale \textit{Alpha}, hypothétique).
    \begin{enumerate}
        \item Expliquez pourquoi la genèse cyclonique est extrêmement défavorable dans le Golfe de Guinée (cisaillement vertical, vorticité relative, ITCZ) malgré des SST chaudes. \textit{(1 pt)}
        \item Si un système se formait à \SI{5}{\degree}N, \SI{5}{\degree}E et dérivait vers le nord sous l'effet du flux de direction (vent moyen \SI{700}{hPa} \approx \SI{10}{kt} du Sud-Ouest), estimez le temps avant impact sur la côte camerounaise (latitude \SI{4}{\degree}N). Quelles mesures de protection \textbf{spécifiques} (au-delà de l'alerte météo) doivent être préconisées pour les populations côtières vulnérables (Douala, Kribi) face à la surcote et aux pluies diluviennes ? \textit{(2 pts)}
    \end{enumerate}
\end{enumerate}
\end{exercice}

\begin{exercice}[Changement climatique : Réorganisation des grands flux]
\label{exo:cc_flux}
Les simulations CMIP6 projettent une modification structurelle de la circulation générale sous forçage anthropique (SSP5-8.5).

\begin{enumerate}
    \item \textbf{Élargissement de la ceinture tropicale.} Les observations et modèles montrent une expansion poleward de la cellule de Hadley et du jet subtropical ($\approx \SI{0.5}{\degree}$ à \SI{1.0}{\degree} par décennie).
    \begin{enumerate}
        \item Reliez cet élargissement à la modification du gradient de température méridien (amplification arctique vs réchauffement tropical) et à la hauteur de la tropopause. \textit{(1 pt)}
        \item Conséquences hydroclimatiques : Expliquez le mécanisme "les zones sèches deviennent plus sèches, les zones humides plus humides" (rich-get-richer) en zone subtropicale descendante (ex: Méditerranée, Afrique australe) via l'advection d'humidité. \textit{(1 pt)}
    \end{enumerate}

    \item \textbf{Ralentissement de la circulation méridienne de renversement atlantique (AMOC).} Un affaiblissement de l'AMOC (\SI{-15}{\%} depuis 1950, projection \SI{-30}{\%} à \SI{-50}{\%} à 2100) est projeté.
    \begin{enumerate}
        \item Identifiez le processus physique principal responsable de ce ralentissement (eau douce, stratification, convection profonde). \textit{(1 pt)}
        \item Analysez les impacts régionaux contrastés : (a) Niveau de la mer sur la côte Est des USA, (b) Position de l'ITCZ atlantique et pluviométrie sahélienne, (c) Stockage de chaleur océanique et rétroaction sur le réchauffement global. \textit{(2 pts)}
    \end{enumerate}

    \item \textbf{Synthèse prospective.} En croisant l'élargissement tropical et le ralentissement de l'AMOC, proposez un scénario cohérent pour l'évolution du risque cyclone (fréquence, intensité, latitude maximale d'intensification) dans l'Atlantique Nord d'ici 2100. Justifiez par les ingrédients thermodynamiques et dynamiques. \textit{(2 pts)}
\end{enumerate}
\end{exercice}
