\coursec{Les acides nucléiques : ADN et ARN}

Au marché Mokolo de Yaoundé, une mère montre fièrement sa fille : « Elle a les yeux de son père et le teint de ma mère ! » Comment des caractères peuvent-ils se transmettre fidèlement de génération en génération, alors que chaque enfant naît d'une seule cellule, l'œuf ? Et pourquoi le nombre de chromosomes ne double-t-il pas à chaque génération ? Chez un couple dont l'enfant présente une trisomie 21, que s'est-il passé au niveau cellulaire ? Cette leçon explique comment la cellule conserve et transmet son information génétique sans en altérer la quantité.

\courssub{À la recherche du support de l'hérédité : expériences historiques}

Avant de connaître la structure de l'ADN, les scientifiques ont dû identifier quelle molécule portait l'information héréditaire. Trois expériences majeures ont levé le doute.

\begin{experience}[L'expérience de Griffith (1928) : le « principe transformant »]
\textbf{Matériel :} Souris, deux souches de pneumocoques (bactéries responsables de pneumonies) :
\begin{itemize}
    \item Souche S (lisse) : capsule de polysaccharides, \textbf{virulente} (tue la souris).
    \item Souche R (rugueuse) : sans capsule, \textbf{non virulente} (souris survit).
\end{itemize}
\textbf{Protocole et observations :}
\begin{enumerate}
    \item Souche S vivante $\rightarrow$ souris meurt.
    \item Souche R vivante $\rightarrow$ souris vit.
    \item Souche S \textbf{tuée à la chaleur} $\rightarrow$ souris vit.
    \item Souche S tuée à la chaleur \textbf{+} Souche R vivante $\rightarrow$ \textbf{souris meurt} ; on retrouve des bactéries S vivantes dans son sang.
\end{enumerate}
\textbf{Interprétation :} Un « principe transformant » issu des bactéries S mortes a transformé les bactéries R vivantes en bactéries S virulentes. L'information de la virulence a été transmise.
\legende{Expérience de Griffith : la transformation de la souche R non virulente en souche S virulente par un facteur issu de la souche S tuée.}
\end{experience}

\begin{experience}[L'expérience d'Avery, MacLeod et McCarty (1944) : l'ADN est le principe transformant]
\textbf{Idée :} Isoler le principe transformant. Les extraits de souche S tuée sont traités par des enzymes spécifiques :
\begin{itemize}
    \item Protéases (détruisent les protéines) $\rightarrow$ transformation \textbf{possible}.
    \item ARNases (détruisent l'ARN) $\rightarrow$ transformation \textbf{possible}.
    \item \textbf{ADNases (détruisent l'ADN)} $\rightarrow$ transformation \textbf{impossible}.
\end{itemize}
\textbf{Conclusion :} Le principe transformant est l'\textbf{ADN}.
\legende{Expérience d'Avery : seule la destruction de l'ADN empêche la transformation.}
\end{experience}

\begin{experience}[L'expérience de Hershey et Chase (1952) : confirmation par le bactériophage]
\textbf{Contexte :} Les bactériophages (virus infectant les bactéries) sont constitués d'une capside protéique et d'ADN interne. Seule l'information génétique entre dans la bactérie.
\textbf{Marquage radioactif :}
\begin{itemize}
    \item Lot 1 : Phages cultivés avec \textsuperscript{32}P (marque l'\textbf{ADN}, le phosphore n'est pas dans les protéines).
    \item Lot 2 : Phages cultivés avec \textsuperscript{35}S (marque les \textbf{protéines}, le soufre n'est pas dans l'ADN).
\end{itemize}
\textbf{Protocole :} Infection de bactéries $\rightarrow$ agitation violente (mélangeur) pour détacher les capsides vides $\rightarrow$ centrifugation (pellet = bactéries, surnageant = capsides).
\textbf{Résultats :}
\begin{itemize}
    \item Lot 1 (\textsuperscript{32}P) : Radioactivité dans le \textbf{pellet} (bactéries).
    \item Lot 2 (\textsuperscript{35}S) : Radioactivité dans le \textbf{surnageant} (capsides).
\end{itemize}
\textbf{Conclusion :} C'est l'ADN, et non la protéine, qui pénètre dans la bactérie et dirige la fabrication de nouveaux phages. L'ADN est le support de l'information génétique.
\legende{Schéma simplifié de l'expérience de Hershey et Chase. Seule l'ADN marqué au \textsuperscript{32}P entre dans la bactérie.}
\end{experience}

\begin{definition}[Acide nucléique]
Une \textbf{macromolécule biologique} formée de l'enchaînement de nombreux \textbf{nucléotides} liés entre eux. Il en existe deux types principaux : l'\textbf{ADN} (acide désoxyribonucléique) et l'\textbf{ARN} (acide ribonucléique). L'ADN est le support universel de l'information génétique chez tous les êtres vivants.
\end{definition}

\courssub{Localisation des acides nucléiques dans la cellule}

\begin{propriete}[Localisation]
\begin{itemize}
    \item \textbf{ADN :} Majoritairement dans le \textbf{noyau} (organisé en chromosomes), aussi dans les mitochondries et les chloroplastes (ADN extranucléaire).
    \item \textbf{ARN :} Synthétisé dans le noyau (transcription), puis exporté vers le \textbf{cytoplasme} (ARNm, ARNt) ou présent dans le \textbf{nucléole} (ARNr, assemblés en ribosomes).
\end{itemize}
\end{propriete}

\courssub{Le nucléotide : l'unité de base}

\begin{definition}[Nucléotide]
Un nucléotide est l'unité monomère des acides nucléiques. Il est composé de \textbf{trois éléments} :
\begin{enumerate}
    \item Un \textbf{groupe phosphate} ($\ce{PO4^{3-}}$).
    \item Un \textbf{sucre à 5 atomes de carbone} (pentose) : \textbf{désoxyribose} dans l'ADN, \textbf{ribose} dans l'ARN.
    \item Une \textbf{base azotée} (cycle carboné portant de l'azote).
\end{enumerate}
\end{definition}

\begin{aretenir}[Les quatre bases azotées]
\begin{itemize}
    \item \textbf{Purines} (double cycle) : \textbf{Adénine (A)}, \textbf{Guanine (G)} — présentes dans l'ADN \textbf{et} l'ARN.
    \item \textbf{Pyrimidines} (cycle simple) :
    \begin{itemize}
        \item \textbf{Cytosine (C)} — présente dans l'ADN \textbf{et} l'ARN.
        \item \textbf{Thymine (T)} — \textbf{uniquement dans l'ADN}.
        \item \textbf{Uracile (U)} — \textbf{uniquement dans l'ARN} (remplace la thymine).
    \end{itemize}
\end{itemize}
\end{aretenir}

\courssub{Structure de l'ADN : la double hélice}

Grâce aux clichés de diffraction X de \textbf{Rosalind Franklin} (1952), \textbf{Watson et Crick} proposent en 1953 le modèle de la double hélice.

\begin{propriete}[Modèle de Watson et Crick (1953)]
L'ADN est constitué de \textbf{deux brins} enroulés en \textbf{double hélice} (hélice droite, pas de 3,4 nm, 10 paires de bases par tour).
\begin{itemize}
    \item Les deux brins sont \textbf{antiparallèles} : l'un dans le sens $5' \rightarrow 3'$, l'autre $3' \rightarrow 5'$.
    \item Le \textbf{squelette} (sucre-phosphate) est à l'extérieur ; les \textbf{bases} sont à l'intérieur, empilées.
    \item \textbf{Complémentarité des bases (règle de Chargaff) :}
    \begin{itemize}
        \item \textbf{A} s'apparie avec \textbf{T} par \textbf{2 liaisons hydrogène}.
        \item \textbf{G} s'apparie avec \textbf{C} par \textbf{3 liaisons hydrogène}.
    \end{itemize}
    \item La séquence d'un brin détermine celle de l'autre (brin codant / brin non codant).
\end{itemize}
\end{propriete}

\begin{methode}[Calculer les pourcentages de bases (règle de Chargaff)]
Les données de Chargaff montrent que dans l'ADN double brin : \textbf{\%A = \%T} et \textbf{\%G = \%C}. Donc \textbf{\%A + \%T + \%G + \%C = 100\%}.
\textbf{Étapes :}
\begin{enumerate}
    \item Identifier la base dont le pourcentage est connu.
    \item Appliquer \%A = \%T et \%G = \%C.
    \item Calculer les pourcentages restants : \%G + \%C = 100\% - (\%A + \%T).
    \item Répartir équitablement : \%G = \%C = (\%G + \%C) / 2.
\end{enumerate}
\textbf{Exemple :} Si \%A = 30\%, alors \%T = 30\%. Reste 40\% pour G+C, donc \%G = \%C = 20\%.
\end{methode}

\begin{exoresolu}[Calcul de composition en bases d'un ADN]
\textbf{Énoncé :} Un fragment d'ADN double brin contient 22\% de cytosine (C). Déterminer les pourcentages des trois autres bases.
\tcblower
\textbf{Solution :}
\begin{enumerate}
    \item \%C = 22\% $\Rightarrow$ \%G = 22\% (règle \%G = \%C).
    \item \%C + \%G = 44\%.
    \item \%A + \%T = 100\% - 44\% = 56\%.
    \item \%A = \%T = 56\% / 2 = \textbf{28\% chacun}.
\end{enumerate}
\textbf{Bilan :} \%A = 28\% ; \%T = 28\% ; \%G = 22\% ; \%C = 22\%.
\end{exoresolu}

\begin{attention}[Piège : Thymine vs Uracile]
\emph{Ne confondez jamais :}
\begin{itemize}
    \item \textbf{Thymine (T)} : base de l'\textbf{ADN} (porte un groupe méthyle \ce{-CH3} sur le cycle).
    \item \textbf{Uracile (U)} : base de l'\textbf{ARN} (pas de groupe méthyle).
\end{itemize}
L'ARN ne contient \textbf{jamais} de thymine (sauf rares ARNt modifiés post-transcription). Au Bac, écrire « l'ARN contient de la thymine » est une erreur éliminatoire.
\end{attention}

\courssub{Structure de l'ARN : un brin simple aux rôles variés}

Contrairement à l'ADN, l'ARN est généralement \textbf{monobrin} (simple brin), plus court, et existe sous plusieurs types fonctionnels.

\begin{tabularx}{\linewidth}{|l|X|X|}\hline
\rowcolor{popblueL}
\textbf{Caractéristique} & \textbf{ADN} & \textbf{ARN} \\ \hline
\textbf{Sucre} & Désoxyribose (H sur C2') & Ribose (OH sur C2') \\ \hline
\textbf{Bases} & A, T, G, C & A, U, G, C \\ \hline
\textbf{Structure} & Double brin, double hélice & Simple brin (peut s'enrouler sur lui-même) \\ \hline

\textbf{Rôle principal} & Stockage stable de l'information & Expression de l'information (transcription, traduction) \\ \hline
\textbf{Types} & Un seul type (ADN génomique) & \textbf{ARNm} (messager), \textbf{ARNt} (transfert), \textbf{ARNr} (ribosomal) \\ \hline
\end{tabularx}
\end{center}

\begin{definition}[Les trois principaux ARN]
\begin{itemize}
    \item \textbf{ARN messager (ARNm) :} Copie d'un gène, porte le code génétique du noyau vers le ribosome (codons).
    \item \textbf{ARN de transfert (ARNt) :} Adaptateur en forme de trèfle, porte un acide aminé spécifique et possède un \textbf{anticodon} complémentaire du codon sur l'ARNm.
    \item \textbf{ARN ribosomal (ARNr) :} Composant structural et catalytique des ribosomes (usines à protéines).
\end{itemize}
\end{definition}

\begin{savaistu}[Rosalind Franklin (1920--1958) : l'ombre de la double hélice]
C'est la cristallographe britannique \textbf{Rosalind Franklin} qui, au King's College de Londres, a obtenu le célèbre \textbf{cliché photographique n°51} (diffraction X de l'ADN hydraté), révélant sans ambiguïté la structure hélicale, le pas de l'hélice et le diamètre constant. Ses données, communiquées sans son accord à Watson et Crick, ont été décisives pour leur modèle de 1953. Elle meurt d'un cancer à 37 ans, quatre ans avant l'attribution du prix Nobel de physiologie ou médecine 1962 à Watson, Crick et Wilkins. Le Nobel ne se décernant pas à titre posthume, elle n'a jamais été honorée pour cette découverte fondamentale.
\end{savaistu}

\begin{exercice}[Entraînement : composition en bases et nature de l'acide nucléique]
\begin{enumerate}
    \item Un ADN double brin contient 18\% d'adénine. Calculer les pourcentages de T, G et C.
    \item Un échantillon d'acide nucléique présente la composition suivante : A = 25\%, U = 25\%, G = 30\%, C = 20\%. S'agit-il d'ADN ou d'ARN ? Justifier.
    \item Pourquoi la règle de Chargaff (\%A = \%T, \%G = \%C) ne s'applique-t-elle pas à l'ARN monobrin ?
\end{enumerate}
\end{exercice}

\begin{corrige}[Exercice n]
\begin{enumerate}
    \item \%A = 18\% $\Rightarrow$ \%T = 18\%. Reste 64\% pour G+C $\Rightarrow$ \%G = \%C = 32\%.
    \item Présence d'\textbf{uracile (U)} et absence de thymine (T) $\Rightarrow$ c'est de l'\textbf{ARN}. De plus, \%A $\neq$ \%U et \%G $\neq$ \%C, compatible avec un brin simple.
    \item L'ARN est monobrin : les bases ne sont pas appariées deux à deux de façon systématique, donc pas de contrainte d'égalité entre A et U ni entre G et C.
\end{enumerate}
\end{corrige}

\coursec{La duplication de l'ADN}

Avant toute division cellulaire — qu'il s'agisse d'une mitose pour la croissance ou d'une méiose pour la formation des gamètes —, la cellule fait face à un impératif absolu : \textbf{chaque cellule fille doit recevoir une copie intégrale et fidèle de l'information génétique}. Puisque l'ADN est le support de cette information, sa quantité doit doubler avant la division. C'est le rôle de la \textbf{duplication de l'ADN} (aussi appelée \textbf{réplication}), que nous allons comprendre en nous appuyant sur des observations expérimentales fondatrices.

\courssub{Le problème posé : doubler l'ADN avant de se diviser}

Rappelons qu'en phase \textbf{$G_1$} du cycle cellulaire, une cellule diploïde humaine contient 46 chromosomes, soit une quantité d'ADN notée \textbf{Q} (ou 2C). Si cette cellule se divisait sans préparer sa division, chaque fille n'hériterait que de la moitié de l'information (Q/2) : l'espèce ne pourrait pas se perpétuer. L'observation au microscope, couplée à des dosages biochimiques, montre qu'avant la mitose, la quantité d'ADN passe de \textbf{Q à 2Q}.

\begin{methode}[Interpréter une courbe de quantité d'ADN]
\begin{enumerate}
    \item \textbf{Repérer les paliers :} Un palier bas correspond à la quantité Q (phase $G_1$), un palier haut à 2Q (phase $G_2$).
    \item \textbf{Identifier la pente montante :} Elle signale la phase S (synthèse de l'ADN). C'est le moment où la duplication a lieu.
    \item \textbf{Identifier la chute brutale :} Elle correspond à la division cellulaire (mitose + cytokinèse) où le noyau se partage en deux.
    \item \textbf{Conclure :} La duplication précède toujours la division ; elle n'a \textbf{jamais} lieu pendant la mitose.
\end{enumerate}
\end{methode}

\begin{attention}[Piège classique au Bac]
Ne confondez \textbf{jamais} \emph{duplication de l'ADN} (phase S, interphase, noyau intact) et \emph{mitose} (phase M, division du noyau, chromosomes condensés). La duplication est la \textbf{préparation} ; la mitose est le \textbf{partage}. Dire « l'ADN se duplique pendant la mitose » est une erreur éliminatoire.
\end{attention}

\courssub{La preuve expérimentale : l'expérience de Meselson et Stahl (1958)}

Comment l'ADN se duplique-t-il ? Trois modèles étaient envisagés dans les années 1950 :
\begin{itemize}
    \item \textbf{Conservatif :} La double hélice parentale reste intacte ; une copie entièrement neuve est synthétisée à côté.
    \item \textbf{Dispersif :} Les deux brins parentaux se fragmentent et se mélangent aux brins neufs dans les deux molécules filles.
    \item \textbf{Semi-conservatif :} Chaque brin parental sert de matrice pour un brin neuf. Chaque molécule fille contient un brin ancien et un brin neuf.
\end{itemize}

Matthew Meselson et Franklin Stahl ont tranché en 1958 grâce à une expérience d'une élégance remarquable, utilisant le \textbf{marquage isotopique} et la \textbf{centrifugation en gradient de densité}.

\begin{experience}[L'expérience de Meselson et Stahl — Démonstration de la réplication semi-conservative]
\textbf{Matériel :} Bactéries \emph{Escherichia coli}, milieu de culture contenant de l'azote lourd (\ce{^{15}N}), puis milieu normal (\ce{^{14}N}), tubes de centrifugation, chlorure de césium (CsCl), lumière UV.
\par\textbf{Protocole :}
\begin{enumerate}
    \item \textbf{Marquage « lourd » :} Culture de bactéries pendant plusieurs générations sur milieu \ce{^{15}N}. Tout l'ADN devient « lourd » (densité élevée). Prélèvement \textbf{Génération 0}.
    \item \textbf{Transfert sur milieu « léger » (\ce{^{14}N}) :} Les bactéries se divisent en incorporant de l'azote léger.
    \item \textbf{Prélèvements successifs :} Après une division (Génération 1), après deux divisions (Génération 2).
    \item \textbf{Centrifugation en gradient de densité :} L'ADN extrait est centrifugé longuement dans du CsCl. Les molécules d'ADN migrent jusqu'à leur position d'équilibre (bande) selon leur densité.
\end{enumerate}
\textbf{Observations (bandes observées au fond du tube) :}
\begin{itemize}
    \item \textbf{Génération 0 (milieu \ce{^{15}N}) :} Une seule bande \textbf{lourde} (bas du tube).
    \item \textbf{Génération 1 (1 division sur \ce{^{14}N}) :} Une seule bande à position \textbf{intermédiaire}.
    \item \textbf{Génération 2 (2 divisions sur \ce{^{14}N}) :} \textbf{Deux bandes} : une \textbf{intermédiaire}, une \textbf{légère} (haut du tube), d'intensité égale.
\end{itemize}
\textbf{Interprétation :}
\begin{itemize}
    \item La bande unique intermédiaire en G1 élimine le modèle \textbf{conservatif} (qui aurait donné deux bandes : lourde + légère) et le modèle \textbf{dispersif} (qui aurait donné une bande intermédiaire, mais qui aurait continué à s'éclaircir uniformément par la suite).
    \item L'apparition de deux bandes (intermédiaire + légère) en G2, en proportions égales, correspond \textbf{exactement} au modèle \textbf{semi-conservatif} : les molécules intermédiaires (1 brin \ce{^{15}N} + 1 brin \ce{^{14}N}) se séparent ; chaque brin \ce{^{15}N} sert de matrice pour un brin \ce{^{14}N} (donnant une molécule intermédiaire), chaque brin \ce{^{14}N} sert de matrice pour un brin \ce{^{14}N} (donnant une molécule légère).
\end{itemize}
\end{experience}

\begin{propriete}[Réplication semi-conservative de l'ADN (Meselson \& Stahl, 1958)]
La duplication de l'ADN suit un mode \textbf{semi-conservatif} : chaque molécule fille est constituée d'un \textbf{brin parental} (conservé) et d'un \textbf{brin néoformé} (synthétisé neuf). Cela assure la fidélité de la transmission de l'information génétique, car chaque brin sert de matrice pour la synthèse de son complémentaire.
\end{propriete}

\courssub{Présentation générale du mécanisme de duplication}

La réplication est un processus enzymatique complexe, mais sa logique est simple : \textbf{ouvrir, copier, refermer}. Elle mobilise une « usine » moléculaire : le \textbf{replisome}.

\begin{enumerate}
    \item \textbf{Initiation (Ouverture) :} Des protéines initiatrices repèrent des séquences spécifiques appelées \textbf{origines de réplication}. L'hélicase « dézippe » la double hélice en brisant les liaisons hydrogène entre bases complémentaires. Deux \textbf{fourches de réplication} s'éloignent dans les deux sens. Des protéines SSB stabilisent les brins simples. La primase pose une amorce d'ARN pour démarrer la synthèse.
    \item \textbf{Élargissement (Synthèse) :} L'enzyme clé est l'\textbf{ADN polymérase}. Elle ne sait faire qu'une chose : \textbf{ajouter des nucléotides complémentaires} au brin matrice, dans le sens $5' \to 3'$.
    \begin{itemize}
        \item Sur le \textbf{brin conducteur} (brin matrice orienté $3' \to 5'$ vers la fourche), la synthèse est \textbf{continue}.
        \item Sur le \textbf{brin retardé} (brin matrice orienté $5' \to 3'$ vers la fourche), la synthèse est \textbf{discontinue} : elle produit de courts fragments (fragments d'Okazaki) qui seront ensuite soudés par l'\textbf{ADN ligase} après remplacement des amorces d'ARN par de l'ADN.
    \end{itemize}
    \item \textbf{Terminaison :} Les fourches se rencontrent. Les derniers nicks sont scellés par la ligase. On obtient deux molécules d'ADN identiques.
\end{enumerate}

\begin{definition}[Duplication (ou réplication) de l'ADN]
Processus par lequel une molécule d'ADN double brin est copiée pour produire \textbf{deux molécules d'ADN identiques} à l'originale, chacune composée d'un brin parental et d'un brin néoformé (mode semi-conservatif). Elle a lieu pendant la \textbf{phase S de l'interphase}, avant toute division cellulaire (mitose ou méiose).
\end{definition}

\courssub{Fidélité, mutations et ordre de grandeur}

La fidélité de la copie est extraordinaire. L'ADN polymérase possède une activité de \textbf{relecture (proofreading)} : elle vérifie chaque base ajoutée et excise l'erreur si la complémentarité n'est pas respectée (A-T, G-C). Le taux d'erreur résiduel est d'environ \textbf{1 erreur pour $10^9$ (un milliard) de nucléotides incorporés}.

\begin{exemplebox}[Chiffres vertigineux : la réplication chez l'humain]
Une cellule humaine diploïde contient environ \textbf{6 milliards de paires de bases} ($6 \times 10^9$). La phase S dure environ \textbf{\SIrange{6}{8}{\hour}}.
\begin{itemize}
    \item Vitesse de polymérisation : $\sim \SI{50}{nt/s}$ par fourche.
    \item Nombre de fourches actives simultanément : \textbf{20 000 à 50 000} (origines de réplication multiples).
    \item Erreurs non corrigées : $\approx$ \textbf{quelques unités à quelques dizaines par division cellulaire}.
\end{itemize}
Ces rares erreurs non corrigées sont des \textbf{mutations}. Elles sont la source ultime de la variabilité génétique (sujet de la leçon sur l'évolution), mais peuvent aussi causer des maladies génétiques ou des cancers si elles touchent des gènes critiques.
\end{exemplebox}

\begin{exoresolu}[Analyse d'un document : Cinétique de réplication]
\textbf{Énoncé :} On suit l'incorporation de thymidine radioactive (\ce{^3H}-thymidine) dans l'ADN de cellules cultivées. La radioactivité de l'ADN extrait est mesurée à différents temps après l'ajout du traceur. Les résultats sont :
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}\hline
Temps (min) & 0 & 10 & 20 & 30 & 40 \\ \hline
Radioactivité (cpm/µg ADN) & 0 & 120 & 480 & 1080 & 1200 \\ \hline
\end{tabular}
\end{center}
\textbf{Questions :}
\begin{enumerate}
    \item Tracer la courbe d'incorporation.
    \item Déduire la durée de la phase S si le plateau correspond à la fin de la réplication.
    \item Expliquer pourquoi la courbe n'est pas linéaire au début.
\end{enumerate}
\tcblower
\textbf{Solution :}
\begin{enumerate}
    \item \textit{Courbe :} Point (0,0), montée rapide puis ralentissement vers un plateau à 1200 cpm/µg. (Forme sigmoïde typique).
    \item \textit{Durée phase S :} Le plateau est atteint vers 40 min. La phase S dure \textbf{environ 40 minutes} dans ces conditions expérimentales.
    \item \textit{Non-linéarité initiale :} Au tout début, peu de fourches de réplication sont actives (initiation progressive aux origines). Le nombre de fourches croît, puis devient maximal (pente maximale vers 20 min), avant de décroître quand les fourches se rencontrent (fin de réplication). C'est la signature cinétique de l'initiation multiple et asynchrone des origines.
\end{enumerate}
\end{exoresolu}

\courssub{Moment de la duplication dans le cycle cellulaire : la phase S}

Il est crucial de situer cet événement dans le temps cellulaire.

\begin{propriete}[Localisation temporelle de la duplication]
La duplication de l'ADN a lieu exclusivement pendant la \textbf{phase S (Synthèse) de l'interphase}.
\begin{itemize}
    \item \textbf{Avant la mitose :} Cycle : $G_1 \to$ \textbf{S (duplication)} $\to G_2 \to$ M (mitose). La cellule entre en mitose avec 2Q d'ADN (chromosomes à deux chromatides).
    \item \textbf{Avant la méiose :} Cycle pré-méiotique : $G_1 \to$ \textbf{S (duplication unique)} $\to$ Méiose I $\to$ Méiose II. \textbf{Une seule duplication pour deux divisions successives.} C'est ce qui permet le passage de 2n (diploïdie) à n (haploïdie).
\end{itemize}
\end{propriete}

\begin{aretenir}[Bilan : La duplication, clé de la pérennité]
\begin{itemize}
    \item \textbf{But :} Doubler la quantité d'ADN (Q $\to$ 2Q) pour que chaque cellule fille reçoive un génome complet.
    \item \textbf{Moment :} Phase S de l'interphase (jamais pendant la division).
    \item \textbf{Mode :} \textbf{Semi-conservatif} (prouvé par Meselson \& Stahl, 1958). Chaque molécule fille = 1 brin parent + 1 brin neuf.
    \item \textbf{Mécanisme :} Ouverture par hélicase $\to$ Synthèse complémentaire par ADN polymérase (sens $5'\to3'$) $\to$ Soudure par ligase.
    \item \textbf{Fidélité :} Extrême ($< 10^{-9}$ erreurs/base). Erreurs résiduelles = mutations.
\end{itemize}
\end{aretenir}

\begin{exercice}[Entraînement : Duplication et cycle cellulaire]
\begin{enumerate}
    \item Une cellule en $G_1$ contient \SI{2}{pg} d'ADN. Quelle sera sa quantité d'ADN en $G_2$ ? Au début de la mitose ? Dans chaque cellule fille après cytokinèse ?
    \item Sur le schéma de l'expérience de Meselson et Stahl (figure ci-dessus), on répète l'expérience jusqu'à la \textbf{Génération 3}. Combien de bandes observe-t-on ? Quelles sont leurs positions relatives et leurs intensités relatives ? Justifiez.
    \item Pourquoi la synthèse du brin retardé est-elle discontinue alors que celle du brin conducteur est continue ? (Indice : sens de lecture de la polymérase vs sens d'avancement de la fourche).
    \item Vrai ou Faux ? Justifiez brièvement.
    \begin{itemize}
        \item « La duplication de l'ADN a lieu pendant la prophase de la mitose. »
        \item « Après réplication, un chromosome est constitué de deux chromatides identiques. »
        \item « L'ADN polymérase synthétise l'ADN dans le sens $3' \to 5'$. »
    \end{itemize}
\end{enumerate}
\end{exercice}

\begin{corrige}[Exercice : Duplication et cycle cellulaire]
\begin{enumerate}
    \item $G_1$ = \SI{2}{pg} (2C). $G_2$ = \textbf{\SI{4}{pg}} (4C, chromosomes à 2 chromatides). Début mitose = \textbf{\SI{4}{pg}}. Cellules filles = \textbf{\SI{2}{pg}} chacune (retour à 2C).
    \item Génération 3 : \textbf{Deux bandes}. Une \textbf{intermédiaire} (intensité 1/4 du total), une \textbf{légère} (intensité 3/4 du total). Justification : Les molécules intermédiaires (G2) se séparent $\to$ 1/2 donnent intermédiaires, 1/2 donnent légères. Les molécules légères (G2) ne donnent que des légères. Total intermédiaires = 1/4, légères = 3/4.
    \item L'ADN polymérase ne lit le brin matrice que dans le sens $3' \to 5'$ et synthétise le brin neuf en $5' \to 3'$. La fourche s'ouvre dans un sens unique. Un brin matrice est orienté $3' \to 5'$ vers la fourche (brin conducteur : synthèse continue dans le sens de la fourche). L'autre brin matrice est orienté $5' \to 3'$ vers la fourche (brin retardé) : la polymérase doit synthétiser « en arrière » par fragments (Okazaki) pour respecter le sens $5' \to 3'$.
    \item \textbf{Faux.} Phase S (interphase). \textbf{Vrai.} (Définition de chromosome dupliqué). \textbf{Faux.} Synthèse $5' \to 3'$ (lecture matrice $3' \to 5'$).
\end{enumerate}
\end{corrige}

\begin{savaistu}[Le Cameroun et la recherche sur l'ADN]
Au Centre de Recherche sur les Maladies Infectieuses (CRMI) de l'Organisation de Coordination pour la lutte contre les Endémies en Afrique Centrale (OCEAC) à Yaoundé, les biologistes moléculaires utilisent quotidiennement la \textbf{PCR (Polymérase Chain Reaction)}. Cette technique, inventée par Kary Mullis (Prix Nobel 1993), n'est rien d'autre qu'une \textbf{duplication de l'ADN in vitro} : on imite la réplication naturelle (dénaturation, hybridation d'amorces, élongation par une ADN polymérase thermostable \emph{Taq}) pour amplifier une séquence d'ADN spécifique. C'est l'outil de base pour diagnostiquer le \textbf{VIH/Sida} (charge virale), le \textbf{Paludisme} (détection \emph{Plasmodium}), la \textbf{Tuberculose} ou encore identifier les variétés améliorées de \textbf{manioc} ou de \textbf{maïs} résistantes aux maladies. Comprendre la duplication de l'ADN, c'est comprendre la médecine et l'agronomie modernes au Cameroun.
\end{savaistu}

\coursec{La biosynthèse des protéines}

Dans la section précédente, nous avons vu comment la cellule \cle{duplique} son ADN avant chaque division, de sorte que chaque cellule fille reçoive une copie intégrale du programme génétique. Mais une question demeure : à quoi sert cet ADN au quotidien ? L'ADN n'est pas une fin en soi : il contient des \emph{instructions} qui doivent être \emph{exécutées}. Or le résultat visible de ces instructions, ce sont les \cle{protéines} : l'hémoglobine qui transporte l'oxygène dans ton sang, la kératine de tes cheveux, les enzymes qui digèrent ton repas, la mélanine qui colore ta peau. Cette section explique comment la cellule passe de l'information écrite dans l'ADN à la fabrication concrète des protéines : c'est la \cle{biosynthèse des protéines}.

\courssub{Le problème posé : une information enfermée dans le noyau}

\begin{exemplebox}[Une usine dont les plans restent au bureau]
Imagine un chantier de construction à Yaoundé. Les plans de l'immeuble sont précieux : on les conserve soigneusement dans le bureau de l'architecte, et il est hors de question de les emporter sur le chantier où ils seraient déchirés ou salis. Comment fait-on ? On réalise une \emph{copie} du plan, qu'un ouvrier emporte sur le chantier, et c'est cette copie que les maçons suivent pour assembler les briques.
\end{exemplebox}

La cellule fonctionne exactement ainsi, et c'est là le problème fondamental :

\begin{itemize}
\item l'\cle{ADN}, support de l'information génétique, est enfermé dans le \cle{noyau} (chez les cellules eucaryotes) et n'en sort jamais ;
\item or les \cle{protéines} sont fabriquées dans le \cle{cytoplasme}, au niveau de petites structures appelées \cle{ribosomes}.
\end{itemize}

Comment l'information franchit-elle donc la distance entre le noyau et le cytoplasme ? La réponse, admise depuis les travaux des années 1960, constitue le \emph{schéma central de la biologie moléculaire} :

\[ \text{ADN} \xrightarrow{\ \text{transcription (noyau)}\ } \text{ARN} \xrightarrow{\ \text{traduction (cytoplasme)}\ } \text{protéine} \]

L'ARN joue le rôle de la \emph{copie de travail} du plan : il est synthétisé dans le noyau à partir de l'ADN, puis il migre dans le cytoplasme où il sert de modèle à la fabrication de la protéine.

\begin{definition}[Gène]
Un \cle{gène} est un segment d'ADN portant l'information nécessaire à la synthèse d'une protéine (ou d'une chaîne polypeptidique). Chaque gène correspond donc à une « recette » précise : le gène de l'hémoglobine, le gène de l'insuline, etc.
\end{definition}

\courssub{Étape 1 : la transcription, une copie dans le noyau}

\begin{definition}[Transcription]
La \cle{transcription} est la synthèse d'une molécule d'\cle{ARN messager} ($ARN_m$) dont la séquence de nucléotides est la copie fidèle d'un gène. Elle a lieu dans le \cle{noyau}, sous l'action d'une enzyme : l'\cle{ARN polymérase}.
\end{definition}

Le mécanisme repose sur la \cle{complémentarité des bases}, exactement comme la duplication de l'ADN, avec une différence : dans l'ARN, la base \emph{uracile} (U) remplace la thymine (T). Les couples de bases sont donc :

\begin{center}
\begin{tabular}{|c|c|}
\hline
\rowcolor{popblueL} \textbf{Base sur l'ADN} & \textbf{Base incorporée dans l'$ARN_m$} \\
\hline
A (adénine) & U (uracile) \\
\hline
T (thymine) & A (adénine) \\
\hline
G (guanine) & C (cytosine) \\
\hline
C (cytosine) & G (guanine) \\
\hline
\end{tabular}
\end{center}

\begin{exemplebox}[Un exemple de transcription]
Si le brin d'ADN lu porte la séquence TAC--GGA--ACT, l'$ARN_m$ synthétisé portera la séquence complémentaire AUG--CCU--UGA. L'$ARN_m$, beaucoup plus court que la molécule d'ADN entière, quitte ensuite le noyau par les pores de l'enveloppe nucléaire et gagne le cytoplasme.
\end{exemplebox}

\courssub{Étape 2 : la traduction, la fabrication au ribosome}

\begin{definition}[Traduction]
La \cle{traduction} est la synthèse d'une protéine (chaîne polypeptidique) à partir de la séquence de l'$ARN_m$. Elle a lieu dans le \cle{cytoplasme}, au niveau des \cle{ribosomes}. Des molécules d'\cle{ARN de transfert} ($ARN_t$) apportent une à une les \cle{acides aminés} correspondant au message.
\end{definition}

L'$ARN_m$ est lu par groupes de \textbf{trois bases} appelés \cle{codons}. Chaque codon désigne un acide aminé précis : c'est le \cle{code génétique}. Le ribosome se déplace le long de l'$ARN_m$ ; à chaque codon, un $ARN_t$ porteur du bon acide aminé vient se fixer, et l'acide aminé s'ajoute à la chaîne en construction. La chaîne polypeptidique grandit ainsi jusqu'à un \emph{codon stop} qui signale la fin de la protéine.

\begin{attention}[Piège classique : la localisation]
L'erreur la plus fréquente au Baccalauréat est de situer la traduction dans le noyau. Retiens bien : la \textbf{transcription} a lieu dans le \textbf{noyau} (produit : $ARN_m$), la \textbf{traduction} a lieu dans le \textbf{cytoplasme}, au niveau des \textbf{ribosomes} (produit : protéine). L'$ARN_m$ est le seul à franchir la frontière entre les deux.
\end{attention}

\courssub{Le code génétique : un dictionnaire universel}

\begin{definition}[Code génétique]
Le \cle{code génétique} est la correspondance entre chaque \cle{codon} (triplet de 3 bases de l'$ARN_m$) et l'\cle{acide aminé} qu'il désigne. Il existe $4^3 = 64$ codons possibles pour 20 acides aminés : le code est dit \emph{redondant} (plusieurs codons peuvent désigner le même acide aminé). Le codon AUG code la méthionine et sert de signal de \emph{départ} ; les codons UAA, UAG et UGA sont des codons \emph{stop}.
\end{definition}

\begin{center}
\begin{tabular}{|c|c|c|c|}
\hline
\rowcolor{popblueL} \textbf{Codon ($ARN_m$)} & \textbf{Acide aminé} & \textbf{Codon ($ARN_m$)} & \textbf{Acide aminé} \\
\hline
AUG & Méthionine (départ) & GAA / GAG & Glutamate \\
\hline
CCU & Proline & GUA / GUG & Valine \\
\hline
UUU & Phénylalanine & UAA / UAG / UGA & Stop (fin) \\
\hline
\end{tabular}
\end{center}

\begin{methode}[Traduire une séquence d'$ARN_m$ en chaîne d'acides aminés]
\begin{enumerate}
\item \textbf{Découper} la séquence de l'$ARN_m$ en triplets (codons) à partir du codon de départ AUG, sans chevauchement.
\item \textbf{Lire} dans le tableau du code génétique l'acide aminé correspondant à chaque codon.
\item \textbf{Enchaîner} les acides aminés dans l'ordre de lecture : c'est la séquence de la protéine.
\item \textbf{S'arrêter} dès qu'un codon stop (UAA, UAG ou UGA) est rencontré.
\end{enumerate}
\emph{Exemple} : AUG--CCU--GAA donne Méthionine -- Proline -- Glutamate.
\end{methode}

\begin{savaistu}[Un code quasi universel]
Le code génétique est pratiquement le même chez la bactérie, le palmier, le moustique et l'Homme : le codon UUU désigne la phénylalanine partout ! Cette universalité est l'une des plus belles preuves de la \emph{parenté de tous les êtres vivants}, descendants d'un ancêtre commun. C'est aussi elle qui rend possible les biotechnologies : on peut faire fabriquer de l'insuline humaine par des bactéries, car elles « lisent » le gène humain avec le même dictionnaire.
\end{savaistu}

\courssub{Une preuve expérimentale : le marquage radioactif des acides aminés}

\begin{experience}[Où sont fabriquées les protéines ? (savoir admis)]
\textbf{Protocole} : on fournit à des cellules en culture des acides aminés marqués par un atome radioactif, pendant un temps très court (quelques secondes), puis on suit la localisation de la radioactivité au fil du temps.
\textbf{Observations} : la radioactivité apparaît d'abord au niveau des \emph{ribosomes} du cytoplasme, puis elle se retrouve ensuite dans d'autres compartiments de la cellule.
\textbf{Interprétation} : les acides aminés sont d'abord \emph{incorporés} au niveau des ribosomes, qui sont donc bien le \emph{lieu de fabrication des protéines}. Les protéines ne migrent qu'ensuite vers leur lieu d'action. Cette expérience confirme que la traduction se déroule dans le cytoplasme, et non dans le noyau.
\end{experience}

\courssub{De la protéine au caractère observable : l'exemple de la drépanocytose}

Pourquoi cette leçon importe-t-elle pour ta santé et ta vie de citoyen ? Parce qu'elle explique \emph{comment une minuscule erreur dans l'ADN devient une maladie visible}. La \cle{drépanocytose} (ou anémie falciforme), maladie génétique fréquente en Afrique subsaharienne et donc au Cameroun, en est l'exemple le plus célèbre.

\begin{experience}[Raisonnement : de la mutation à la maladie]
\begin{enumerate}
\item \textbf{Observation} : les personnes atteintes de drépanocytose possèdent des globules rouges déformés, en forme de \emph{faucille}, qui transportent mal l'oxygène et obstruent les petits vaisseaux (crises douloureuses, anémie).
\item \textbf{Analyse biochimique} : leur hémoglobine diffère de l'hémoglobine normale par \emph{un seul acide aminé} sur les 146 de la chaîne : à la 6\textsuperscript{e} position, le \emph{glutamate} est remplacé par la \emph{valine}.
\item \textbf{Analyse génétique} : ce remplacement provient de la mutation d'\emph{un seul nucléotide} du gène : le codon GAG (glutamate) devient GUG (valine).
\end{enumerate}
\textbf{Conclusion} : une mutation ponctuelle de l'ADN $\rightarrow$ un codon modifié $\rightarrow$ un acide aminé changé $\rightarrow$ une protéine anormale $\rightarrow$ un caractère observable (la maladie). La protéine est bien le \emph{trait d'union} entre le gène et le caractère.
\end{experience}

\begin{propriete}[Relation gène -- protéine -- caractère]
Le caractère observable d'un individu (groupe sanguin, couleur des yeux, maladie héréditaire) résulte de l'action de \cle{protéines}, dont la structure est dictée par les \cle{gènes}. Toute modification de la séquence d'un gène peut modifier la protéine et donc le caractère. Autres exemples : les \emph{groupes sanguins} A, B et O dépendent d'\emph{enzymes} qui modifient la surface des globules rouges ; la \emph{couleur des yeux} dépend des protéines intervenant dans la fabrication de la mélanine de l'iris.
\end{propriete}

\begin{exoresolu}[Traduction d'un mini-gène]
Énoncé. Un brin d'ADN porte la séquence : TAC -- CCA -- CTT -- ATT. 1) Donner la séquence de l'$ARN_m$ transcrit. 2) Traduire cet $ARN_m$ en chaîne d'acides aminés à l'aide du tableau du code génétique simplifié (on donne en plus : GGU = glycine). 3) Que se passe-t-il si le 7\textsuperscript{e} nucléotide de l'ADN (le C du codon CTT) est remplacé par un A ?
\tcblower
Solution détaillée.
1) \textbf{Transcription} (complémentarité A--U, T--A, G--C, C--G) :
\[ \text{ADN : TAC -- CCA -- CTT -- ATT} \;\longrightarrow\; \text{$ARN_m$ : AUG -- GGU -- GAA -- UAA} \]
2) \textbf{Traduction} : AUG = méthionine (départ) ; GGU = glycine ; GAA = glutamate ; UAA = codon \emph{stop}. La chaîne obtenue est donc : Méthionine -- Glycine -- Glutamate, puis arrêt.
3) Le codon CTT devient ATT, donc sur l'$ARN_m$ le codon GAA (glutamate) devient UAA, un \emph{codon stop}. La traduction s'interrompt prématurément : la protéine est \emph{raccourcie} et le plus souvent non fonctionnelle. Cet exercice illustre comment une mutation d'un seul nucléotide peut bouleverser une protéine entière, comme dans la drépanocytose.
\end{exoresolu}

1) Recopie et complète : « La transcription a lieu dans le \ldots{} et produit un \ldots{} ; la traduction a lieu dans le \ldots{}, au niveau des \ldots{}, et produit une \ldots{}. »
2) À partir de l'$ARN_m$ AUG--UUU--GUG--UAG, donne la séquence d'acides aminés obtenue.
3) Explique en trois phrases maximum pourquoi on dit que « la protéine est le trait d'union entre le gène et le caractère », en t'appuyant sur l'exemple de la drépanocytose.
\end{exercice}

\begin{corrige}[Exercice : À toi de jouer]
1) La transcription a lieu dans le \textbf{noyau} et produit un \textbf{$ARN_m$} ; la traduction a lieu dans le \textbf{cytoplasme}, au niveau des \textbf{ribosomes}, et produit une \textbf{protéine} (chaîne polypeptidique).
2) AUG = méthionine ; UUU = phénylalanine ; GUG = valine ; UAG = stop. La chaîne est : Méthionine -- Phénylalanine -- Valine.
3) Le gène porte l'information de fabrication de la protéine ; la protéine (ici l'hémoglobine) assure une fonction visible (transport de l'oxygène par les globules rouges). Si le gène est muté, la protéine est anormale et le caractère change : globules en faucille et maladie. La protéine relie donc l'information génétique au caractère observable.
\end{corrige}

\begin{aretenir}[L'essentiel de la biosynthèse des protéines]
\begin{itemize}
\item Schéma central : \textbf{ADN} $\xrightarrow{\text{transcription (noyau)}}$ \textbf{$ARN_m$} $\xrightarrow{\text{traduction (cytoplasme, ribosomes)}}$ \textbf{protéine}.
\item La \textbf{transcription} copie un gène en $ARN_m$ par complémentarité des bases (enzyme : ARN polymérase).
\item La \textbf{traduction} lit l'$ARN_m$ codon par codon ; les $ARN_t$ apportent les acides aminés selon le \textbf{code génétique}, universel.
\item Un \textbf{gène} = un segment d'ADN codant une protéine ; une protéine détermine un \textbf{caractère} (exemple : drépanocytose, GAG $\to$ GUG).
\end{itemize}
\end{aretenir}

\coursec{La méiose : de la diploïdie à l'haploïdie}

Après avoir vu comment l'information génétique s'exprime pour fabriquer nos protéines (\emph{biosynthèse des protéines}), nous devons comprendre comment cette information est transmise d'une génération à l'autre sans être diluée ni doublée. Comment l'espèce humaine conserve-t-elle ses 46 chromosomes à chaque génération, alors que la fécondation fusionne deux cellules ? C'est le rôle de la \cle{méiose}, une division cellulaire très particulière que nous allons décortiquer.

\courssub{Le problème : maintenir le nombre de chromosomes constant}

Observons les caryotypes humains. Une cellule somatique (peau, foie, sang) possède \textbf{46 chromosomes} organisés en 23 paires : c'est une cellule \cle{diploïde} (notée \textbf{2n}). Les chromosomes d'une même paire sont \cle{homologues} (même taille, même position du centromère, mêmes gènes), l'un vient du père, l'autre de la mère.

Les gamètes (spermatozoïdes, ovocytes) ne possèdent que \textbf{23 chromosomes} : ce sont des cellules \cle{haploïdes} (notées \textbf{n}). Ils n'ont qu'un seul chromosome par paire.

\begin{attention}[Piège : la mitose ne suffit pas pour faire des gamètes]
Si les gamètes étaient produits par \textbf{mitose} (division conservant 2n), un spermatozoïde (2n) féconderait un ovocyte (2n) pour donner un zygote \textbf{4n}. À la génération suivante : 8n, puis 16n... Le nombre de chromosomes doublerait à chaque génération ! La méiose est la solution : elle \textbf{divise par deux} le nombre de chromosomes avant la fécondation.
\end{attention}

\begin{definition}[Cellules diploïdes et haploïdes]
\begin{itemize}
    \item \textbf{Cellule diploïde (2n)} : possède ses chromosomes par paires homologues (2 exemplaires de chaque chromosome). C'est le cas des cellules somatiques.
    \item \textbf{Cellule haploïde (n)} : possède un seul exemplaire de chaque chromosome (pas de paires homologues). C'est le cas des gamètes.
\end{itemize}
\end{definition}

\courssub{La méiose : deux divisions successives pour une seule duplication}

La méiose est une \textbf{double division cellulaire} qui transforme une cellule diploïde (2n) en \textbf{quatre cellules haploïdes (n)}. Elle ne comporte qu'\textbf{une seule réplication de l'ADN} (phase S), suivie de deux divisions : la \textbf{méiose I} (réductionnelle) et la \textbf{méiose II} (équationnelle).

\begin{propriete}[Principe général de la méiose]
\begin{itemize}
    \item \textbf{Entrée} : 1 cellule diploïde (2n, 2C ADN en G1, 4C après réplication).
    \item \textbf{Sortie} : 4 cellules haploïdes (n, 1C ADN chacune).
    \item \textbf{Deux divisions successives} : Méiose I puis Méiose II, sans réplication d'ADN entre les deux.
\end{itemize}
\end{propriete}

\courssub{La première division : la méiose réductionnelle (2n $\rightarrow$ n)}

C'est l'étape clé. Elle sépare les \textbf{chromosomes homologues} (les chromosomes paternels d'un côté, les maternels de l'autre).

\begin{enumerate}
    \item \textbf{Prophase I (longue et cruciale)} : Les chromosomes homologues s'apparient deux par deux pour former des \cle{bivalents} (ou tétades : 4 chromatides). C'est le moment du \textbf{crossing-over} : échange de morceaux d'ADN entre chromatides non-sœurs d'homologues $\rightarrow$ \textbf{brassage intrachromosomique}.
    \item \textbf{Métaphase I} : Les bivalents s'alignent sur la plaque équatoriale. L'orientation de chaque paire est \textbf{aléatoire} (père à gauche ou à droite) $\rightarrow$ \textbf{brassage interchromosomique} (répartition aléatoire des homologues).
    \item \textbf{Anaphase I} : Les chromosomes \textbf{homologues se séparent} et migrent vers les pôles opposés. \textbf{Les chromatides sœurs restent collées} au centromère. C'est ici que le nombre de chromosomes (compté par centromères) passe de \textbf{2n à n}.
    \item \textbf{Télophase I} : Deux cellules filles se forment, chacune avec n chromosomes (mais chaque chromosome est encore double = 2 chromatides).
\end{enumerate}

\courssub{La deuxième division : la méiose équationnelle (n $\rightarrow$ n)}

Elle ressemble à une mitose : elle sépare les \textbf{chromatides sœurs}. Il n'y a \textbf{pas de réplication d'ADN} avant.

\begin{enumerate}
    \item \textbf{Prophase II} : Condensation des chromosomes (déjà doubles).
    \item \textbf{Métaphase II} : Chromosomes (doubles) alignés à l'équateur.
    \item \textbf{Anaphase II} : Les \textbf{chromatides sœurs se séparent} au niveau du centromère $\rightarrow$ deviennent des chromosomes filles (simples).
    \item \textbf{Télophase II} : 4 cellules haploïdes (n) formées, chacune avec des chromosomes simples (1 chromatide).
\end{enumerate}

\begin{methode}[Comment distinguer méiose I et méiose II sur un schéma ?]
\begin{enumerate}
    \item Compte le nombre de chromosomes (centromères) dans la cellule mère.
    \item Regarde ce qui se sépare à l'anaphase :
    \begin{itemize}
        \item Si ce sont des \textbf{chromosomes entiers} (formés de 2 chromatides) qui s'éloignent $\rightarrow$ \textbf{Méiose I (Réductionnelle)}. Le nombre de chromosomes diminue (ex: 4 $\rightarrow$ 2).
        \item Si ce sont des \textbf{chromatides sœurs} (bras d'un même chromosome) qui s'éloignent $\rightarrow$ \textbf{Méiose II (Équationnelle)}. Le nombre de chromosomes (centromères) ne change pas (ex: 2 $\rightarrow$ 2), mais ils deviennent simples.
    \end{itemize}
\end{enumerate}
\end{methode}

\begin{attention}[Erreur fréquente au Bac]
La réduction du nombre de chromosomes (passage de 2n à n) a lieu à l'\textbf{Anaphase I} (séparation des homologues), \textbf{pas} à l'Anaphase II. À l'Anaphase II, on sépare des chromatides : le nombre de chromosomes (compté par centromères) reste n.
\end{attention}

\courssub{Le brassage génétique : source de diversité}

La méiose ne se contente pas de réduire le nombre de chromosomes, elle \textbf{mélange les allèles} pour créer des gamètes uniques. Deux mécanismes majeurs :

\begin{enumerate}
    \item \textbf{Crossing-over (Prophase I)} : Échange de segments d'ADN entre chromatides non-sœurs d'un même bivalent. Crée de nouvelles combinaisons d'allèles \textbf{sur un même chromosome} (brassage intrachromosomique).
    \item \textbf{Répartition aléatoire des homologues (Métaphase I/Anaphase I)} : Chaque paire d'homologues s'oriente indépendamment des autres sur la plaque équatoriale. Pour 23 paires, il y a $2^{23}$ combinaisons possibles de chromosomes paternels/maternels dans un gamète (brassage interchromosomique).
\end{enumerate}

\begin{savaistu}[La loterie génétique humaine]
Grâce au seul brassage interchromosomique ($2^{23}$), un seul homme peut produire théoriquement \textbf{plus de 8 millions} ($8\,388\,608$) de types de spermatozoïdes différents rien qu'en mélangeant ses chromosomes paternels et maternels. Avec le crossing-over, le nombre de combinaisons possibles est quasi infini. C'est pourquoi deux frères (vrais jumeaux exceptés) sont toujours génétiquement uniques !
\end{savaistu}

\begin{itemize}

    \item Après Méiose I : \textbf{2Q} (n chromosomes doubles).
    \item Après Méiose II : \textbf{Q} (n chromosomes simples). C'est le gamète.
\end{itemize}
\end{aretenir}

La méiose produit des gamètes haploïdes (n). La \textbf{fécondation} fusionne un spermatozoïde (n) et un ovocyte (n) pour reformer un \textbf{zygote diploïde (2n)}. Le cycle est bouclé : la constance du nombre de chromosomes de l'espèce est assurée.

\courssub{Quand la méiose déraille : la non-disjonction et la trisomie 21}

Parfois, lors de l'anaphase I (homologues) ou de l'anaphase II (chromatides), les chromosomes ne se séparent pas correctement : c'est la \cle{non-disjonction}. Un gamète se retrouve avec un chromosome en trop (n+1) ou en moins (n-1).

Si un gamète (n+1) fusionne avec un gamète normal (n) $\rightarrow$ Zygote \textbf{2n+1} = \textbf{Trisomie}.

\begin{exemplebox}[La trisomie 21 (Syndrome de Down)]
C'est la trisomie la plus fréquente (environ 1 naissance sur 700 au Cameroun comme ailleurs). Le chromosome 21 est présent en \textbf{3 exemplaires} au lieu de 2. Le caryotype s'écrit \textbf{47,XX,+21} (fille) ou \textbf{47,XY,+21} (garçon).
\begin{itemize}
    \item \textbf{Origine} : Non-disjonction le plus souvent en Méiose I chez la mère (ovogenèse), fréquence liée à l'âge maternel.
    \item \textbf{Conséquences} : Retard mental modéré, traits morphologiques caractéristiques, risque cardiaque, mais une vie sociale et scolaire possible avec une prise en charge adaptée (écoles inclusives, orthophonie, suivi médical en centres hospitaliers comme à Yaoundé ou Douala).
\end{itemize}
\end{exemplebox}

\begin{exoresolu}[Analyse d'un caryotype et détermination de l'anomalie méiotique]
\textbf{Énoncé :} On observe le caryotype d'un fœtus obtenu par amniocentèse. Le chromosome 21 est présent en trois exemplaires. Les deux chromosomes 21 d'origine maternelle portent des allèles différents pour un marqueur génétique proche du centromère (allèles $M_1$ et $M_2$). Le chromosome 21 d'origine paternelle porte l'allèle $M_3$. Le fœtus possède les trois allèles $M_1$, $M_2$ et $M_3$.
\textbf{Question :} La non-disjonction a-t-elle eu lieu lors de la méiose I ou de la méiose II chez la mère ? Justifier.

\tcblower
\textbf{Solution :}
\begin{enumerate}
    \item \textbf{Rappel} : En Méiose I, ce sont les \textbf{homologues} qui se séparent (allèles différents $M_1$ et $M_2$). En Méiose II, ce sont les \textbf{chromatides sœurs} (identiques génétiquement, sauf crossing-over récent près du centromère) qui se séparent.
    \item \textbf{Observation} : La mère a transmis \textbf{deux chromosomes 21 différents} ($M_1$ et $M_2$). Ce sont ses deux homologues.
    \item \textbf{Conclusion} : Les deux homologues maternels n'ont \textbf{pas été séparés}. Ils sont restés ensemble dans l'ovocyte. C'est une non-disjonction en \textbf{Anaphase I (Méiose I)}.
    \item \textit{Note} : Si c'était une non-disjonction en Méiose II, la mère aurait transmis deux chromatides sœurs $\rightarrow$ deux chromosomes 21 \textbf{identiques} (même allèle $M_1$ ou $M_2$).
\end{enumerate}
\end{exoresolu}

\begin{exercice}[Entraînement : Quantité d'ADN et nombre de chromosomes]
Une cellule animale diploïde possède 2n = 8 chromosomes en phase G1.
\begin{enumerate}
    \item Combien de chromosomes et de chromatides cette cellule contient-elle au début de la méiose I (après phase S) ?
    \item Combien de chromosomes et de chromatides dans chacune des deux cellules à la fin de la méiose I ?
    \item Combien de chromosomes et de chromatides dans chacun des quatre gamètes à la fin de la méiose II ?
    \item Quelle est la quantité d'ADN (en Q) à chaque étape ci-dessus ?
\end{enumerate}
\end{exercice}

\begin{corrige}[Exercice n°1]
\begin{enumerate}
    \item \textbf{Début Méiose I (G2)} : 2n = 8 chromosomes \textbf{doubles} $\rightarrow$ 8 chromosomes, 16 chromatides. ADN = \textbf{4Q}.
    \item \textbf{Fin Méiose I} : Séparation homologues. 2 cellules \textbf{n = 4 chromosomes doubles} $\rightarrow$ 4 chromosomes, 8 chromatides par cellule. ADN = \textbf{2Q}.
    \item \textbf{Fin Méiose II} : Séparation chromatides. 4 cellules \textbf{n = 4 chromosomes simples} $\rightarrow$ 4 chromosomes, 4 chromatides par cellule. ADN = \textbf{Q}.
    \item Résumé ADN : G1 (2Q) $\xrightarrow{\text{Phase S}}$ 4Q $\xrightarrow{\text{Méiose I}}$ 2Q $\xrightarrow{\text{Méiose II}}$ Q.
\end{enumerate}
\end{corrige}

\courssub{Synthèse : La méiose, gardienne de l'identité de l'espèce}

\begin{aretenir}[Bilan de la section]
\begin{itemize}
    \item La \textbf{méiose} est une double division (I réductionnelle, II équationnelle) précédée d'une seule duplication de l'ADN.
    \item Elle assure le passage de la \textbf{diploïdie (2n) à l'haploïdie (n)} en séparant les chromosomes homologues (Anaphase I) puis les chromatides sœurs (Anaphase II).
    \item Elle génère de la \textbf{diversité génétique} par crossing-over (brassage intrachromosomique) et répartition aléatoire des homologues (brassage interchromosomique).
    \item La \textbf{fécondation} (fusion de deux gamètes n) rétablit la diploïdie (2n) : le nombre de chromosomes de l'espèce est constant de génération en génération.
    \item Une \textbf{non-disjonction} (échec de séparation) conduit à des gamètes anormaux (n+1 ou n-1) et, après fécondation, à des anomalies chromosomiques comme la \textbf{trisomie 21}.
\end{itemize}
\end{aretenir}

\courssub{Le bilan du parcours de l'information génétique}

Après avoir étudié la duplication de l'ADN, la biosynthèse des protéines et la méiose, nous pouvons maintenant relier ces processus pour comprendre comment l'identité biologique se transmet de génération en génération. L'information génétique circule dans un sens privilégié : \textbf{ADN $\rightarrow$ ARN $\rightarrow$ Protéine}.

    \item \textbf{Transcription} (dans le noyau) : un brin d'ADN sert de modèle pour synthétiser un \cle{ARN messager (ARNm)}.

\end{itemize}
\end{propriete}

\begin{propriete}[Constante du nombre chromosomique de l'espèce (savoir admis)]
La \textbf{fécondation} (fusion de deux gamètes haploïdes) rétablit la \textbf{diploïdie (2n)} et assure la \textbf{constance du nombre de chromosomes} caractéristique de l'espèce, de génération en génération.
\begin{itemize}
    \item \textbf{Méiose} : passage de 2n à n (réduction). Elle a lieu lors de la formation des gamètes (\cle{gamétogenèse}).
    \item \textbf{Fécondation} : passage de n + n à 2n (restauration). Elle a lieu lors de la rencontre spermatozoïde + ovocyte.
\end{itemize}
Sans méiose, le nombre de chromosomes doublerait à chaque génération ; sans fécondation, il serait divisé par deux.
\end{propriete}

\begin{exemplebox}[Chez l'Homme : une stabilité remarquable]
Chez l'\emph{Homo sapiens}, la formule chromosomique est \textbf{2n = 46 chromosomes} (23 paires).
\begin{itemize}
    \item Cellule somatique (peau, foie, muscle...) : 46 chromosomes (23 paires homologues).
    \item Gamète (spermatozoïde ou ovocyte) : 23 chromosomes (une seule copie de chaque paire).
    \item Zygote (après fécondation) : 46 chromosomes (23 paires reconstituées : 1 paternelle + 1 maternelle).
\end{itemize}
Ce chiffre \textbf{46} est maintenu constant depuis des centaines de milliers de générations grâce à l'alternance méiose/fécondation.
\end{exemplebox}

\courssub{Mitose et méiose : deux divisions complémentaires}

La mitose et la méiose ne sont pas en concurrence ; elles ont des rôles biologiques distincts et complémentaires au sein de l'organisme.

\begin{methode}[Construire un tableau comparatif Mitose / Méiose]
Pour ne rien oublier au Bac, suis ces 5 critères obligatoires :
\begin{enumerate}
    \item \textbf{Contexte / Localisation} : Où cela se passe-t-il ? (Cellules somatiques vs Gonades).
    \item \textbf{Nombre de divisions successives} : 1 pour la mitose, 2 pour la méiose (I et II).
    \item \textbf{Nombre de cellules filles produites} : 2 pour la mitose, 4 pour la méiose.
    \item \textbf{État chromosomique des cellules filles} : Identiques à la mère (2n $\rightarrow$ 2n) vs Réduites de moitié (2n $\rightarrow$ n).
    \item \textbf{Rôle biologique} : Croissance, renouvellement, réparation (mitose) vs Reproduction sexuée, brassage génétique (méiose).
\end{enumerate}
\end{methode}

\courssub{Fidélité de la transmission et diversité des individus}

Comment l'espèce reste-t-elle « elle-même » alors que chaque individu est unique ? La réponse tient en deux mots : \textbf{fidélité} et \textbf{brassage}.

\begin{aretenir}[L'équilibre stabilité / diversité]
\begin{itemize}
    \item \textbf{Stabilité de l'espèce (Fidélité)} : La \textbf{duplication de l'ADN} est semi-conservative et très précise (enzymes de relecture). Elle assure que le « programme » de base ne change pas (ou très peu) d'une génération à l'autre. C'est la condition de la pérennité de l'espèce.
    \item \textbf{Diversité des individus (Brassage)} : La \textbf{méiose} (\cle{brassage intrachromosomique} / \cle{crossing-over} + \cle{brassage interchromosomique} / \cle{ségrégation aléatoire}) et la \textbf{fécondation} (rencontre aléatoire de deux gamètes parmi des millions) créent des combinaisons génétiques uniques à chaque individu.
\end{itemize}
C'est cette tension entre \textbf{conservation} (même nombre de chromosomes, mêmes gènes au même locus) et \textbf{innovation} (allèles différents, nouveaux génotypes) qui permet à l'espèce de s'adapter sans perdre son identité.
\end{aretenir}

\courssub{Ouverture : quand le mécanisme déraille}

Les mécanismes que nous venons d'étudier sont d'une précision remarquable, mais ils ne sont pas infaillibles. Les « accidents » de la duplication ou de la méiose sont à l'origine de la variabilité... et parfois de pathologies.

\begin{attention}[Piège majeur : Quantité d'ADN (Q) vs Nombre de chromosomes (n)]
C'est l'erreur n°1 en Terminale ! Ne confonds jamais :
\begin{itemize}
    \item \textbf{Le nombre de chromosomes (n, 2n)} : se compte en \textbf{centromères} (constriction primaire). Il change \textbf{uniquement} en Anaphase de Méiose I (séparation des homologues : 2n $\rightarrow$ n) et à la Fécondation (n + n $\rightarrow$ 2n).
    \item \textbf{La quantité d'ADN (Q, 2Q, 4Q)} : correspond à la \textbf{masse d'ADN}. Elle double lors de la \textbf{duplication (phase S)} (2Q $\rightarrow$ 4Q) et se divise par 2 à \textbf{chaque division} (Mitose : 4Q $\rightarrow$ 2Q ; Méiose I : 4Q $\rightarrow$ 2Q ; Méiose II : 2Q $\rightarrow$ 1Q).
\end{itemize}
\textbf{Astuce :} Sur un graphique, regarde les \textbf{paliers}. Un palier à 4Q = après duplication, avant division. Un palier à 2Q = cellule somatique en G1 ou gamète juste après méiose I. Un palier à 1Q = gamète mature.
\end{attention}

\begin{savaistu}[Mutations et anomalies chromosomiques au Cameroun]
Les erreurs de duplication (mutations ponctuelles) peuvent modifier un allèle : c'est le cas de la \textbf{drépanocytose} (mutation du gène $\beta$-globine, fréquente en zone palustre au Cameroun), qui confère une résistance au paludisme à l'état hétérozygote (AS).
Les erreurs de méiose (non-disjonction) entraînent des anomalies du nombre de chromosomes : \textbf{Trisomie 21} (trois chromosomes 21 au lieu de deux), syndrome de Turner (XO), syndrome de Klinefelter (XXY). Ces anomalies seront étudiées en détail dans les leçons suivantes sur la variabilité et la pathologie génétique.
\end{savaistu}

\begin{exoresolu}[Analyse d'un cycle cellulaire complet]
\emph{Énoncé :} On suit une cellule souche spermatogoniale humaine (2n=46, 2Q ADN) qui subit une mitose, puis l'une des cellules filles entre en méiose pour former des spermatozoïdes. Un spermatozoïde féconde un ovocyte (n=23, 1Q ADN). Indiquer pour chaque étape clé : le nombre de chromosomes (n ou 2n) et la quantité d'ADN (Q).
\tcblower
\emph{Solution détaillée :}
\begin{enumerate}
    \item \textbf{Départ (Spermatogonie G1)} : 2n = 46 chr ; 2Q ADN.
    \item \textbf{Après phase S (Duplication)} : 2n = 46 chr (chaque chr = 2 chromatides) ; \textbf{4Q ADN}.
    \item \textbf{Fin Mitose (2 Spermatogonies)} : 2n = 46 chr ; \textbf{2Q ADN} (chacune).
    \item \textbf{Entrée Méiose I (Spermatocyte I)} : 2n = 46 chr ; 4Q ADN (après nouvelle phase S pré-méiotique).
    \item \textbf{Fin Méiose I (2 Spermatocytes II)} : \textbf{n = 23 chr} (séparation homologues) ; \textbf{2Q ADN} (chaque chr = 2 chromatides).
    \item \textbf{Fin Méiose II (4 Spermatides)} : \textbf{n = 23 chr} (séparation chromatides) ; \textbf{1Q ADN}.
    \item \textbf{Spermatozoïde mature} : \textbf{n = 23 chr} ; \textbf{1Q ADN}.
    \item \textbf{Fécondation (Zygote)} : \textbf{2n = 46 chr} (23 paternels + 23 maternels) ; \textbf{2Q ADN} (1Q + 1Q).
\end{enumerate}
\textbf{Conclusion :} La mitose conserve 2n/2Q. La méiose passe de 2n/4Q $\rightarrow$ n/2Q $\rightarrow$ n/1Q. La fécondation restaure 2n/2Q.
\end{exoresolu}

\courssub{Ressources à mobiliser}
\begin{itemize}
\item \textbf{Duplication de l'ADN :} Semi-conservative. Avant la méiose (phase S), chaque chromosome se réplique $\to$ 2 chromatides sœurs. Quantité d'ADN : $2C \xrightarrow{\text{Phase S}} 4C$.
\item \textbf{Méiose I (Réductionnelle) :} Séparation des \textbf{chromosomes homologues} (paternels vs maternels). Passage $2n \to n$ (mais chromosomes toujours doublés : $4C \to 2C$ en quantité d'ADN par cellule fille). \textit{Anaphase I : migration des chromosomes homologues vers les pôles opposés.}
\item \textbf{Méiose II (Équationnelle) :} Séparation des \textbf{chromatides sœurs}. Mécanisme identique à la mitose. Passage $n$ (doublés) $\to n$ (simples). Quantité d'ADN : $2C \to 1C$ par cellule.
\item \textbf{Non-disjonction :} Échec de la séparation. Anaphase I $\to$ gamètes $n+1$ et $n-1$ (chromosomes doublés). Anaphase II $\to$ gamètes $n+1$, $n-1$, $n$, $n$ (chromosomes simples).
\item \textbf{Expression génique :} Transcription (Noyau) : ADN $\to$ ARNm (remplacement T par U). Traduction (Cytoplasme/Ribosome) : ARNm $\to$ Protéine (code génétique universel, triplet = 1 acide aminé).
\item \textbf{Relation Protéine-Caractère :} La structure primaire (suite d'acides aminés) détermine le repliement 3D et la fonction. Une substitution d'un acide aminé (Glu $\to$ Val) dans l'hémoglobine $\beta$ modifie la solubilité $\to$ polymérisation en faible O$_2$ $\to$ falciformation.
\end{itemize}

\textbf{1. ANALYSE CARYOTYPIQUE (Doc 1)}
\begin{enumerate}
\item \textbf{Formule :} \textbf{47,XY,+21}. Nombre total : \textbf{47 chromosomes} (au lieu de 46). Anomalie : \textbf{Trisomie 21} (nom courant) ou \textbf{trisomie libre du chromosome 21} (nom savant). Présence de \textbf{trois} chromosomes 21 au lieu d'une paire.
\item \textbf{Type :} Anomalie \textbf{autosomique} (affecte les autosomes). Justification : Le chromosome 21 est un autosome (paire 21 sur 22 paires d'autosomes). Les gonosomes (XY) sont normaux (un X, un Y).
\end{enumerate}

\textbf{2. ANALYSE MÉIOTIQUE (Doc 2)}
\begin{enumerate}
\item \textbf{Phase fautive :} \textbf{Anaphase I} (Méiose I).
\textbf{Justification :} Sur la courbe théorique, la quantité d'ADN chute brutalement de \textbf{4C (Métaphase I) à 2C (Télophase I)} lors de l'anaphase I, car les chromosomes homologues (chacun constitué de 2 chromatides) se séparent. Sur la courbe de la mère, cette chute est \textbf{incomplète} : la quantité stagne à \textbf{$\approx$ 2,3C} au lieu de 2C. Cela signifie que les deux chromosomes 21 homologues (chacun 2 chromatides) sont restés ensemble dans la même cellule fille (l'ovocyte), au lieu de migrer vers des pôles opposés. L'ovocyte reçoit donc un excédent d'ADN correspondant à un chromosome 21 supplémentaire (doublé).
\item \textbf{Type de méiose :} \textbf{Méiose I (Réductionnelle)}.
\textbf{Différence clé :}
\begin{itemize}
\item \textbf{Méiose I (Réductionnelle) :} Sépare les \textbf{chromosomes homologues} (père/mère). Réduit le nombre de chromosomes ($2n \to n$). Les chromatides sœurs \textbf{ne se séparent pas} (cohésion centromérique protégée).
\item \textbf{Méiose II (Équationnelle) :} Sépare les \textbf{chromatides sœurs}. Maintient le nombre de chromosomes ($n \to n$). Similaire à une mitose.
\end{itemize}
Ici, l'excédent d'ADN correspond à un chromosome \textbf{entier doublé} (2 chromatides), preuve que ce sont les \textbf{homologues} qui n'ont pas désuni (Anaphase I). Si c'était l'Anaphase II, l'excédent n'aurait concerné qu'une \textbf{seule chromatide} (quantité d'ADN finale $\approx$ 1,15C au lieu de 1C, mais quantité à la fin de Méiose I normale à 2C).
\item \textbf{Rôle de la méiose :} La fécondation fusionne deux gamètes ($n$ chromosomes chacun) pour former le zygote ($2n$). Si la méiose n'existait pas, les gamètes seraient diploïdes ($2n$) $\to$ zygote tétraploïde ($4n$) $\to$ doublement du nombre de chromosomes à chaque génération. La méiose \textbf{réduit de moitié} le nombre de chromosomes (passage diploïdie $\to$ haploïdie) pour compenser le doublement de la fécondation, assurant la \textbf{constance du nombre chromosomique} de l'espèce (46 chez l'humain).
\end{enumerate}

\textbf{3. ANALYSE MOLÉCULAIRE (Doc 3)}
\begin{enumerate}
\item \textbf{Retraduction :}
\begin{align*}
\textbf{Allèle } \beta^{\text{A}} \text{ (Normal)} &: \text{ADN } \texttt{CCT GAG GAG} \\
&\xrightarrow{\text{Transcription (T$\to$U)}} \text{ARNm } \texttt{CCU GAG GAG} \\
&\xrightarrow{\text{Traduction}} \textbf{Pro-Glu-Glu} \quad (\text{Proline - Glutamate - Glutamate}) \\[6pt]
\textbf{Allèle } \beta^{\text{S}} \text{ (Drépanocytaire)} &: \text{ADN } \texttt{CCT GTG GAG} \\
&\xrightarrow{\text{Transcription}} \text{ARNm } \texttt{CCU GUG GAG} \\
&\xrightarrow{\text{Traduction}} \textbf{Pro-Val-Glu} \quad (\text{Proline - Valine - Glutamate})
\end{align*}
\item \textbf{Mutation :} \textbf{Substitution} d'une base unique (ponctuelle) : \textbf{A $\to$ T} (ou U sur l'ARN) au 2\textsuperscript{e} codon (position 6 du codon GAG $\to$ GUG). C'est une \textbf{transversion} (purine A $\to$ pyrimidine T). Conséquence : \textbf{Mutation faux-sens (missense)} : le Glutamate (Glu, acide, polaire, chargé négativement) est remplacé par de la \textbf{Valine (Val, hydrophobe)}.
\item \textbf{Du gène au phénotype :} L'hémoglobine est une protéine tétramère ($\alpha_2\beta_2$). La Valine en position 6 de la chaîne $\beta$ crée une \textbf{tache hydrophobe} à la surface de la protéine. En conditions de faible oxygénation (tissus), l'hémoglobine S (HbS) polymérise en longues fibres rigides qui déforment le globule rouge en \textbf{faucille (drépanocyte)}. Ces cellules rigides bloquent les capillaires (\textbf{vaso-occlusion}) $\to$ ischémie, douleurs crises, anémie hémolytique. Paul est \textbf{homozygote SS} : il ne synthétise \textbf{que} de l'HbS anormale. Ses parents sont \textbf{hétérozygotes AS} : ils produisent 50\% HbA (normale, soluble) et 50\% HbS. L'HbA empêche la polymérisation massive $\to$ porteurs sains (phénotype normal, sauf effort intense/hypoxie).
\end{enumerate}

\textbf{4. SYNTHÈSE ET CONSEIL GÉNÉTIQUE}
\begin{enumerate}
\item \textbf{Jean (Trisomie 21) :} Anomalie \textbf{chromosomique numérique} (excès de matériel génétique : 3 copies du chr 21 au lieu de 2) due à une \textbf{non-disjonction méiotique maternelle (Anaphase I)} $\to$ déséquilibre du dosage génique global (surdosage des gènes du chr 21).
\item \textbf{Paul (Drépanocytose) :} Anomalie \textbf{moléculaire qualitative} (mutation ponctuelle faux-sens Glu$\to$Val sur le gène $\beta$-globine, chr 11) transmise sur \textbf{mode récessif autosomique} $\to$ synthèse d'une protéine structurellement défectueuse (HbS) qui polymérise.
\item \textbf{Distinction :} L'un est un accident de \textbf{ségrégation chromosomique} (quantitatif, aléatoire, non héréditaire), l'autre une \textbf{mutation de séquence ADN} (qualitatif, héréditaire, mendélien).
\item \textbf{Risques pour une grossesse future :}
\begin{itemize}
\item \textbf{Trisomie 21 :} Risque de récidive \textbf{faible ($\approx$ 1\% au-dessus du risque maternel lié à l'âge)}. La non-disjonction est un accident stochastique de la méiose ; elle ne se « transmet » pas (sauf translocation parentale rare, exclue ici car caryotype parental non fourni mais trisomie libre). Le risque de base dépend de l'âge de la mère.
\item \textbf{Drépanocytose :} Parents tous deux \textbf{AS (hétérozygotes)}. Croisement $AS \times AS$ :
\begin{center}
\begin{tabular}{|c|c|c|}\hline
 & \textbf{A} & \textbf{S} \\ \hline
\textbf{A} & AA (Sain) & AS (Porteur sain) \\ \hline
\textbf{S} & AS (Porteur sain) & \textbf{SS (Malade)} \\ \hline
\end{tabular}
\end{center}
Risque \textbf{1/4 (25\%)} à chaque grossesse d'avoir un enfant \textbf{SS (drépanocytaire majeur)}. Risque 1/2 (50\%) porteur sain AS. Risque 1/4 (25\%) sain non porteur AA.
\end{itemize}
\end{enumerate}
\textbf{Conclusion :} Double pathologie génétique indépendante dans la fratrie. Conseil : Dépistage prénatal (ADN libre circulant ou amniocentèse) pour trisomie 21 (risque âge maternel) + diagnostic prénatal ciblé mutation $\beta^{\text{S}}$ (biologie moléculaire sur trophoblaste/liquide amniotique) pour drépanocytose. Prise en charge multidisciplinaire (pédiatrie, hématologie, génétique) recommandée.
\tcblower
\textbf{Commentaire pédagogique :} Ce corrigé modélise la rédaction scientifique attendue : vocabulaire précis (non-disjonction, faux-sens, haploïdie/diploïdie), articulation entre niveaux d'organisation (chromosome $\to$ ADN $\to$ protéine $\to$ phénotype), et raisonnement causal. L'élève doit montrer qu'il distingue l'échelle chromosomique (méiose, caryotype) de l'échelle moléculaire (mutation, code génétique).
\end{exoresolu}

\courssub{Bilan de l'activité}
Cette activité d'intégration a permis de mettre en évidence la \textbf{cohérence multi-échelle} de la génétique :
\begin{itemize}
\item Au niveau \textbf{chromosomal} (Doc 1 \& 2) : La méiose est le \textbf{garde-fou numérique} de l'espèce. Son étape réductionnelle (Anaphase I) est critique : son échec (non-disjonction) engendre des gamètes aneuploïdes ($n+1$ ou $n-1$) et, après fécondation, des syndromes chromosomiques (ici Trisomie 21). La courbe d'ADN est un outil puissant pour \textbf{diagnostiquer l'étape exacte} de l'erreur (I vs II).
\item Au niveau \textbf{moléculaire} (Doc 3) : La fidélité de la \textbf{duplication} et de la \textbf{transcription/traduction} assure la pérennité de l'information. Une seule erreur de copie (mutation ponctuelle) change un acide aminé, modifie la protéine (HbS), et bascule le phénotype (drépanocytose).
\item \textbf{Articulation :} La quantité d'ADN (caryotype, méiose) et la qualité de l'ADN (séquence, mutation) sont les deux piliers de l'identité biologique. L'un assure \textbf{« la bonne dose »} (nombre de chromosomes), l'autre \textbf{« la bonne recette »} (séquence des gènes).
\item \textbf{Compétence citoyenne :} L'élève est désormais capable de lire un caryotype, d'interpréter une courbe cinétique d'ADN, de retraduire une séquence, et de calculer un risque génétique mendélien — bases indispensables pour comprendre le conseil génétique, le dépistage néonatal (drépanocytose au Cameroun) et le diagnostic prénatal (trisomie 21).
\end{itemize}

\begin{methode}[Décrire l'organisation architecturale des acides nucléiques (ADN et ARN)]
    \textbf{Objectif :} Présenter simplement la structure des nucléotides et la différence fondamentale entre \cle{ADN} et \cle{ARN}.
    \begin{enumerate}
        \item \textbf{Identifier les trois composants d'un nucléotide :} une \cle{base azotée}, un \cle{oside} (sucre) et un groupement \cle{phosphate}.
        \item \textbf{Distinguer l'oside :} \emph{désoxyribose} pour l'ADN (pas d'OH en 2'), \emph{ribose} pour l'ARN (OH en 2').
        \item \textbf{Lister les bases azotées :} Adénine (A), Guanine (G), Cytosine (C) communes aux deux ; Thymine (T) spécifique à l'ADN, Uracile (U) spécifique à l'ARN.
        \item \textbf{Décrire l'architecture globale :}
        \begin{itemize}
            \item \textbf{ADN :} Double brin \cle{antiparallèle}, en \cle{double hélice} (Watson-Crick), bases \cle{complémentaires} (A=T, G≡C) liées par \cle{ponts hydrogène}.
            \item \textbf{ARN :} Généralement \cle{simple brin}, peut s'enrouler sur lui-même (structure secondaire), plus instable (OH en 2').
        \end{itemize}
        \item \textbf{Utiliser le vocabulaire précis :} Brin, nucléotide, complémentarité, antiparallèle, 5' $\rightarrow$ 3'.
    \end{enumerate}
    \textbf{Réflexe Bac :} Un tableau comparatif (Nature du sucre, Bases, Nombre de brins, Localisation, Fonction) vaut souvent 2 à 3 points.
\end{methode}

\begin{savaistu}[La drépanocytose au Cameroun : une histoire d'ADN et de protéine]
    Au Cameroun, la \cle{drépanocytose} (maladie génétique héréditaire fréquente) illustre parfaitement le lien ADN $\rightarrow$ Protéine $\rightarrow$ Caractère. Une mutation ponctuelle sur le gène de la \cle{bêta-globine} (chromosome 11) remplace un acide aminé (Acide glutamique $\rightarrow$ Valine) dans l'hémoglobine (HbS au lieu de HbA). Cette modification de la \cle{structure primaire} de la protéine entraîne son agrégation en fibres sous faible oxygène, déformant les globules rouges en "faucilles" (drepanocytes). Ces cellules bloquent les capillaires (crises vaso-occlusives) et s'hémolysent (anémie). La connaissance de cette base moléculaire guide le diagnostic néonatal (dépistage à la naissance dans les maternités de Yaoundé, Douala, etc.) et la prise en charge (hydroxyurée, vaccination anti-pneumocoque, transfusion). C'est la biologie moléculaire au service de la santé publique.
\end{savaistu}

\begin{exoresolu}[Comparaison structurale ADN / ARN]
    \textbf{Énoncé :} Le tableau ci-dessous présente des caractéristiques structurales des acides nucléiques. Complétez-le en cochant les cases correspondantes (une seule réponse par ligne).
    \begin{center}
        \begin{tabularx}{\linewidth}{|l|X|X|}\hline
            \rowcolor{popblueL} \textbf{Caractéristique} & \textbf{ADN} & \textbf{ARN} \\ \hline
            Sucre (oside) & & \\ \hline
            Bases azotées présentes & & \\ \hline
            Nombre de brins (état natif) & & \\ \hline
            Localisation principale dans la cellule eucaryote & & \\ \hline
            Rôle principal & & \\ \hline
        \end{tabularx}
    \end{center}
    \tcblower
    \textbf{Solution détaillée :}
    \begin{enumerate}
        \item \textbf{Sucre :} L'ADN contient du \textbf{désoxyribose} (d'où son nom Acide DésoxyriboNucléique). L'ARN contient du \textbf{ribose}. La différence chimique est la présence d'un groupement -OH sur le carbone 2' du ribose, absent sur le désoxyribose (d'où "désoxy").
        \item \textbf{Bases azotées :} Les deux possèdent les bases puriques (Adénine A, Guanine G) et pyrimidiques (Cytosine C). L'ADN a la \textbf{Thymine (T)}, l'ARN a l'\textbf{Uracile (U)}. On note : A, G, C, T pour l'ADN ; A, G, C, U pour l'ARN.
        \item \textbf{Nombre de brins :} L'ADN est \textbf{double brin} (hélice double). L'ARN est généralement \textbf{simple brin} (même s'il forme des boucles par appariement intra-moléculaire).
        \item \textbf{Localisation :} Chez l'eucaryote (notre cas), l'ADN est essentiellement dans le \textbf{noyau} (et mitochondries/chloroplastes). L'ARN est synthétisé dans le noyau mais \textbf{fonctionne dans le cytoplasme} (ARNm, ARNt, ARNr aux ribosomes).
        \item \textbf{Rôle :} L'ADN est le \textbf{stockage de l'information génétique} (support de l'hérédité). L'ARN est l'\textbf{intermédiaire} (ARNm : copie du gène ; ARNt : adaptateur acides aminés ; ARNr : composant ribosomes) pour la \cle{biosynthèse des protéines}.
    \end{enumerate}
    \begin{center}
        \begin{tabularx}{\linewidth}{|l|X|X|}\hline
            \rowcolor{popblueL} \textbf{Caractéristique} & \textbf{ADN} & \textbf{ARN} \\ \hline
            Sucre (oside) & Désoxyribose & Ribose \\ \hline
            Bases azotées présentes & A, G, C, T & A, G, C, U \\ \hline
            Nombre de brins (état natif) & Double brin & Simple brin \\ \hline
            Localisation principale (eucaryote) & Noyau & Noyau (synthèse) / Cytoplasme (fonction) \\ \hline
            Rôle principal & Stockage information héréditaire & Expression de l'information (synthèse protéines) \\ \hline
        \end{tabularx}
    \end{center}
    \textbf{Conclusion :} L'ADN et l'ARN sont tous deux des polymères de nucléotides, mais leurs différences structurales (sucre, base T/U, brins) dictent leurs fonctions distinctes : stabilité du stockage pour l'ADN, versatilité fonctionnelle pour l'ARN.
\end{exoresolu}

\begin{methode}[Réaliser / Interpréter une maquette simple de molécule d'ADN]
    \textbf{Objectif :} Représenter ou analyser un modèle physique/schématique de la double hélice respectant les règles de \cle{complémentarité} et d'\cle{antiparallélisme}.
    \begin{enumerate}
        \item \textbf{Matériel mental (ou réel) :} Deux brins (ficelles, pailles), 4 couleurs de perles/attaches pour les bases (A, T, G, C), repères 5' et 3'.
        \item \textbf{Assembler les nucléotides :} Enfiler sur chaque brin une suite de bases \emph{aléatoire} (ex: A-T-G-C-C-A). Respecter le sens 5' $\rightarrow$ 3' (repère sur le sucre).
        \item \textbf{Former les paires complémentaires :} Appariement strict \textbf{A = T} (2 ponts H) et \textbf{G ≡ C} (3 ponts H). Vérifier que chaque base d'un brin fait face à sa complémentaire sur l'autre.
        \item \textbf{Vérifier l'antiparallélisme :} Si le brin 1 va de 5' (gauche) $\rightarrow$ 3' (droite), le brin 2 doit aller de 3' (gauche) $\rightarrow$ 5' (droite). Les repères 5' et 3' sont aux extrémités opposées.
        \item \textbf{Twister :} Tordre l'ensemble pour former une \textbf{double hélice droite} (dans le sens des aiguilles d'une montre si on regarde de haut).
    \end{enumerate}
    \textbf{Point clé :} La séquence d'un brin dicte entièrement la séquence de l'autre (information redondante). C'est la base de la \cle{duplication semi-conservative}.
\end{methode}

\begin{exoresolu}[Analyse d'une maquette d'ADN]
    \textbf{Énoncé :} On observe une maquette d'ADN réalisée par un élève. Un brin (brin 1) porte l'étiquette "5'" à son extrémité gauche et "3'" à droite. La séquence de bases lue de gauche à droite est : \textbf{A - T - G - C - C - A}.
    \begin{enumerate}
        \item Déduire la séquence du brin complémentaire (brin 2) en indiquant son sens 5' $\rightarrow$ 3'.
        \item L'élève a représenté les liaisons A-T par 3 traits et G-C par 2 traits. Corrigez cette erreur.
        \item Pourquoi dit-on que les deux brins sont "antiparallèles" ?
    \end{enumerate}
    \tcblower
    \textbf{Solution détaillée :}
    \begin{enumerate}
        \item \textbf{Séquence du brin 2 :}
            \begin{itemize}
                \item Règle de complémentarité : A s'apparie avec T ; G s'apparie avec C.
                \item Brin 1 (5' $\rightarrow$ 3') : \textbf{A - T - G - C - C - A}
                \item Brin 2 (face à face, donc 3' $\rightarrow$ 5' de gauche à droite) : \textbf{T - A - C - G - G - T}
                \item \textbf{Mais on doit donner la séquence dans le sens 5' $\rightarrow$ 3'.} Il faut la lire de droite à gauche sur le schéma ci-dessus.
                \item Brin 2 (5' $\rightarrow$ 3') : \textbf{T - G - G - C - A - T}.
            \end{itemize}
            \textbf{Réponse :} 5' - \textbf{T G G C A T} - 3'.
        \item \textbf{Correction des liaisons hydrogène :}
            \begin{itemize}
                \item Paire \textbf{A = T} : \textbf{2} ponts hydrogène (représentés par 2 traits ou pointillés).
                \item Paire \textbf{G ≡ C} : \textbf{3} ponts hydrogène (représentés par 3 traits).
            \end{itemize}
            L'élève a inversé le nombre. C'est une erreur classique : plus de liaisons = plus stable (température de dénaturation plus élevée pour les zones riches en GC).
        \item \textbf{Antiparallélisme :} Les deux brins ont des sens chimiques opposés. Le carbone 5' (porteur du phosphate) d'un brin fait face au carbone 3' (porteur de l'OH libre) de l'autre brin. Les deux extrémités 5' sont aux bouts opposés de la molécule. Cela est indispensable pour le fonctionnement des enzymes (ADN polymérases) qui ne synthétisent qu'en 5' $\rightarrow$ 3'.
    \end{enumerate}
    \textbf{Conclusion :} Une maquette correcte valide la complémentarité des bases (Chargaff) et l'antiparallélisme, prérequis pour comprendre la duplication.
\end{exoresolu}

\begin{methode}[Identifier les grandes étapes de la duplication de l'ADN (Réplication)]
    \textbf{Objectif :} Restituer le mécanisme \cle{semi-conservatif}, ses acteurs enzymatiques principaux et le résultat génétique.
    \begin{enumerate}
        \item \textbf{Contexte :} Se produit pendant la \textbf{phase S} de l'interphase (cycle cellulaire), avant la mitose ou la méiose. Lieu : \textbf{Noyau}.
        \item \textbf{Étape 1 : Ouverture (Dénaturation locale).} Enzyme \textbf{Hélicase} brise les ponts H $\rightarrow$ \cle{Fourche de réplication}. Protéines SSB stabilisent les brins simples. Topoisomérase (Gyrase) relâche la tension (superenroulement).
        \item \textbf{Étape 2 : Amorçage.} \textbf{Primase} (ARN polymérase) pose un court fragment d'ARN (amorce) en 5' $\rightarrow$ 3' sur chaque brin matrice.
        \item \textbf{Étape 3 : Élongation.} \textbf{ADN Polymérase} ajoute des dNTP complémentaires \textbf{uniquement en 5' $\rightarrow$ 3'}.
            \begin{itemize}
                \item Brin \textbf{conducteur (leading)} : Synthèse continue dans le sens de la fourche.
                \item Brin \textbf{retardé (lagging)} : Synthèse discontinue (\cle{fragments d'Okazaki}) dans le sens opposé.
            \end{itemize}
        \item \textbf{Étape 4 : Finition.} ADN Polymérase I (ou RNase H + FEN1) retire les amorces ARN $\rightarrow$ ADN Polymérase comble les trous $\rightarrow$ \textbf{ADN Ligase} soude les fragments (liaison phosphodiester 3'-OH / 5'-P).
        \item \textbf{Résultat :} 2 molécules d'ADN filles, chacune avec \textbf{1 brin parental (conservé) + 1 brin neuf} $\rightarrow$ \textbf{Duplication semi-conservative} (Meselson \& Stahl).
    \end{enumerate}
    \textbf{Formule bilan :} 1 ADN double brin $\xrightarrow{\text{Phase S}}$ 2 ADN double brin identiques (sauf erreurs rares).
\end{methode}

\begin{exoresolu}[Mécanisme de la réplication et semi-conservation]
    \textbf{Énoncé :} On étudie l'expérience historique de Meselson et Stahl (1958) sur la bactérie \emph{E. coli}.
    \begin{enumerate}
        \item Les bactéries sont cultivées pendant plusieurs générations dans un milieu riche en azote lourd (\ce{^{15}N}). L'ADN est donc "lourd". Elles sont ensuite transférées dans un milieu à azote léger (\ce{^{14}N}). Préciser la densité de l'ADN extrait après :
            \begin{enumerate}
                \item 0 génération (milieu \ce{^{15}N}) ;
                \item 1 génération (milieu \ce{^{14}N}) ;
                \item 2 générations (milieu \ce{^{14}N}).
            \end{enumerate}
        \item Expliquer pourquoi ces résultats valident le modèle \textbf{semi-conservatif} et invalident le modèle conservatif et le modèle dispersif.
    \end{enumerate}
    \tcblower
    \textbf{Solution détaillée :}
    \begin{enumerate}
        \item \textbf{Analyse des densités (centrifugation sur gradient de CsCl) :}
            \begin{itemize}
                \item \textbf{Génération 0 (milieu \ce{^{15}N}) :} Tout l'ADN contient du \ce{^{15}N}. \textbf{Une seule bande : "Lourde" (HL ou LL).}
                \item \textbf{Génération 1 (1 division en \ce{^{14}N}) :} Chaque molécule fille = 1 brin parent (\ce{^{15}N}) + 1 brin neuf (\ce{^{14}N}). Masse intermédiaire. \textbf{Une seule bande : "Intermédiaire" (HL).} \textit{(Modèle conservatif aurait donné 2 bandes : Lourde et Légère ; Dispersif aurait donné 1 bande Intermédiaire aussi).}
                \item \textbf{Génération 2 (2 divisions en \ce{^{14}N}) :} Les molécules HL se dupliquent.
                    \begin{itemize}
                        \item Brin \ce{^{15}N} + brin \ce{^{14}N} neuf $\rightarrow$ Molécule HL (Intermédiaire).
                        \item Brin \ce{^{14}N} (de la gén 1) + brin \ce{^{14}N} neuf $\rightarrow$ Molécule LL (Légère).
                    \end{itemize}
                    \textbf{Deux bandes : 50\% Intermédiaire (HL) et 50\% Légère (LL).} \textit{(Conservatif : 25\% Lourde, 75\% Légère ; Dispersif : 1 bande intermédiaire plus légère que gén 1).}
            \end{itemize}
        \item \textbf{Interprétation :}
            \begin{itemize}
                \item \textbf{Semi-conservatif :} Prédit exactement : Gén 0 (Lourde), Gén 1 (1 bande Intermédiaire), Gén 2 (2 bandes Intermédiaire + Légère). \textbf{Validé.}
                \item \textbf{Conservatif :} Prédit Gén 1 : 2 bandes (Lourde + Légère). \textbf{Observé : 1 bande. Invalide.}
                \item \textbf{Dispersif :} Prédit Gén 2 : 1 bande unique (mélange fin). \textbf{Observé : 2 bandes distinctes. Invalide.}
            \end{itemize}
    \end{enumerate}
    \textbf{Conclusion :} La duplication de l'ADN est semi-conservative : chaque molécule fille conserve un brin parental, assurant la fidélité de la transmission de l'information génétique.
\end{exoresolu}

\begin{methode}[Identifier et situer les grandes étapes de la biosynthèse des protéines]
    \textbf{Objectif :} Relier l'information génétique (ADN) à la protéine fonctionnelle via l'ARNm. Deux étapes majeures : \cle{Transcription} (Noyau) $\rightarrow$ \cle{Traduction} (Cytoplasme/Ribosome).
    \begin{enumerate}
        \item \textbf{TRANSCRIPTION (Noyau) :} "Copie d'un gène".
            \begin{itemize}
                \item \textbf{Acteurs :} ADN (gène = séquence codante), \textbf{ARN Polymérase}, Nucléotides ribonucléotidiques (ATP, UTP, GTP, CTP).
                \item \textbf{Processus :} Ouverture locale $\rightarrow$ Élémentation ARN complémentaire au brin matrice (codogène) : A$\rightarrow$U, T$\rightarrow$A, G$\rightarrow$C, C$\rightarrow$G. Sens 5' $\rightarrow$ 3'.
                \item \textbf{Maturation (Eucaryotes) :} \cle{Épissage} (retrait introns, jonction exons) + Coiffe 5' + Queue poly-A 3' $\rightarrow$ \textbf{ARNm mature}.
                \item \textbf{Sortie :} Pore nucléaire $\rightarrow$ Cytoplasme.
            \end{itemize}
        \item \textbf{TRADUCTION (Cytoplasme - Ribosome) :} "Décodage en protéine".
            \begin{itemize}
                \item \textbf{Acteurs :} ARNm, Ribosomes (ARNr + protéines), \textbf{ARNt} (adaptateurs : anticodon + acide aminé spécifique), Facteurs d'initiation/élongation/termination, ATP/GTP.
                \item \textbf{Code génétique :} 64 codons (triplets ARNm) $\rightarrow$ 20 acides aminés + STOP. Universel, dégénéré, sans chevauchement.
                \item \textbf{Étapes :}
                    \begin{enumerate}
                        \item \textbf{Initiation :} Petit sous-unité ribosome + ARNm (codon AUG initiation) + ARNt Mét (anticodon UAC) $\rightarrow$ Grand sous-unité.
                        \item \textbf{Élargation (cycle) :} 1. Entrée ARNt (site A) codon/anticodon. 2. Liaison peptidique (site P $\rightarrow$ A). 3. Translocation (ribosome avance 1 codon).
                        \item \textbf{Termination :} Codon STOP (UAA, UAG, UGA) reconnu par facteur de libération $\rightarrow$ Hydrolyse liaison peptidyl-ARNt $\rightarrow$ Libération polypeptide + dissociation ribosome.
                    \end{enumerate}
            \end{itemize}
        \item \textbf{Devenir :} Pliage (conformation 3D), modifications post-traductionnelles $\rightarrow$ Protéine fonctionnelle (enzyme, structure, hormone, anticorps...).
    \end{enumerate}
    \textbf{Repères spatiaux :} \textbf{Noyau} = Transcription/Maturation. \textbf{Cytoplasme} = Traduction.
\end{methode}

\begin{exoresolu}[Du gène à la protéine : transcription et traduction]
    \textbf{Énoncé :} Un gène humain codant pour une petite peptide a la séquence codante (brin codant / sens) suivante sur l'ADN :
    \[ \text{5' - ATG GCT TTA CGA - 3'} \]
    \begin{enumerate}
        \item Déterminer la séquence de l'\textbf{ARNm mature} correspondant (après maturation). Indiquer les extrémités 5' et 3'.
        \item En utilisant le tableau du code génétique (fourni ci-dessous), déterminer la séquence d'acides aminés de la protéine finale.
        \item Préciser le rôle des ARNt dans cette étape.
    \end{enumerate}
    \textbf{Extrait du code génétique (ARNm) :}
    \begin{center}
        \begin{tabular}{|c|c|c|c|}\hline
            \rowcolor{popblueL} \textbf{Codon} & \textbf{AA} & \textbf{Codon} & \textbf{AA} \\ \hline
            AUG & Met (Début) & GCU, GCC, GCA, GCG & Ala \\
            UUU, UUC & Phe & CGU, CGC, CGA, CGG & Arg \\
            UAA, UAG, UGA & STOP & \textbf{UUA, UUG} & \textbf{Leu} \\ \hline
        \end{tabular}
    \end{center}
    \tcblower
    \textbf{Solution détaillée :}
    \begin{enumerate}
        \item \textbf{Transcription (ADN $\rightarrow$ ARNm) :}
            \begin{itemize}
                \item Le brin codant (donné) a la même séquence que l'ARNm (sauf T $\rightarrow$ U).
                \item Brin codant ADN : 5' - \textbf{A T G G C T T T A C G A} - 3'
                \item ARNm (complémentaire au brin matrice, donc identique au brin codant T$\rightarrow$U) : 5' - \textbf{A U G G C U U U A C G A} - 3'
            \end{itemize}
            \textbf{Réponse :} 5' - \textbf{AUG GCU UUA CGA} - 3'.
        \item \textbf{Traduction (ARNm $\rightarrow$ Protéine) :} Lecture par triplets (codons) depuis le codon START (AUG) dans le sens 5' $\rightarrow$ 3'.
            \begin{align*}
                \text{Codon 1 : } & \textbf{AUG} \rightarrow \textbf{Méthionine (Met)} \text{ (Start)} \\
                \text{Codon 2 : } & \textbf{GCU} \rightarrow \textbf{Alanine (Ala)} \\
                \text{Codon 3 : } & \textbf{UUA} \rightarrow \textbf{Leucine (Leu)} \quad (\text{Présent dans le tableau : UUA = Leu}) \\
                \text{Codon 4 : } & \textbf{CGA} \rightarrow \textbf{Arginine (Arg)}
            \end{align*}
            \textit{Attention : La séquence donnée ne contient pas de codon STOP. La traduction s'arrêterait au codon STOP suivant dans la séquence réelle.}
            \textbf{Séquence peptidique :} \textbf{Met - Ala - Leu - Arg - ...}
        \item \textbf{Rôle des ARNt :} Ce sont les \textbf{adaptateurs moléculaires}. Chaque ARNt possède :
            \begin{itemize}
                \item Un \textbf{anticodon} (3 bases) complémentaire d'un codon de l'ARNm (ex: UAC pour AUG).
                \item Une extrémité 3' (CCA) où est fixée \textbf{spécifiquement} l'acide aminé correspondant (ici Méthionine pour l'ARNt anticodon UAC).
            \end{itemize}
            Ils assurent la correspondance physique entre le langage nucléique (codon) et le langage protéique (acide aminé).
    \end{enumerate}
    \textbf{Conclusion :} L'information circule ADN $\xrightarrow{\text{Transcription}}$ ARNm $\xrightarrow{\text{Traduction}}$ Protéine. La fidélité repose sur la complémentarité des bases (Transcription) et le code génétique lu par les ARNt (Traduction).
\end{exoresolu}

\begin{methode}[Établir la différence entre méiose réductionnelle (I) et méiose équationnelle (II)]
    \textbf{Objectif :} Comparer les deux divisions successives de la \cle{méiose} pour expliquer le passage 2n $\rightarrow$ n.
    \begin{enumerate}
        \item \textbf{Rappel préliminaire :} Une seule duplication de l'ADN (Phase S pré-méiotique) $\rightarrow$ Chromosomes à 2 chromatides. Cellule de départ : \textbf{Diploïde (2n), 2c ADN $\rightarrow$ 4c ADN après Phase S}.
        \item \textbf{Méiose I (Réductionnelle) :} \textbf{Sépare les chromosomes homologues.}
            \begin{itemize}
                \item \textbf{Prophase I :} \textbf{Point clé.} Appariement homologues (synapse) $\rightarrow$ \textbf{Chiasmes} (croisement visible = \cle{brassage intrachromosomique}). \textbf{Brassage interchromosomique} (ségrégation aléatoire des paires).
                \item \textbf{Métaphase I :} Plaque équatoriale = Paires de chromosomes homologues (bivalents).
                \item \textbf{Anaphase I :} Migration des \textbf{chromosomes entiers} (2 chromatides) vers les pôles. \textbf{Centromères NE se divisent PAS.}
                \item \textbf{Résultat :} 2 cellules \textbf{haploïdes (n)}, chromosomes à \textbf{2 chromatides} (2c ADN par cellule).
            \end{itemize}
        \item \textbf{Méiose II (Équationnelle) :} \textbf{Sépare les chromatides sœurs.} (Similaire à une mitose mais sur cellules n).
            \begin{itemize}
                \item Pas de Phase S intermédiaire (pas de duplication ADN).
                \item \textbf{Anaphase II :} Division des \textbf{centromères} $\rightarrow$ Migration des \textbf{chromatides} (devenus chromosomes filles).
                \item \textbf{Résultat :} 4 cellules \textbf{haploïdes (n)}, chromosomes à \textbf{1 chromatide} (1c ADN par cellule) = \textbf{Gamètes}.
            \end{itemize}
        \item \textbf{Tableau récapitulatif (à connaître par cœur) :}
    \end{enumerate}
    \begin{center}
        \begin{tabularx}{\linewidth}{|l|X|X|}\hline
            \rowcolor{popblueL} \textbf{Critère} & \textbf{Méiose I (Réductionnelle)} & \textbf{Méiose II (Équationnelle)} \\ \hline
            \textbf{But} & 2n $\rightarrow$ n (Réduction nombre chromo) & n $\rightarrow$ n (Séparation chromatides) \\ \hline
            \textbf{Prophase} & Longue, Synapse, \textbf{Chiasmes (Brassage intra)} & Courte, Pas de synapse \\ \hline
            \textbf{Métaphase} & Paires d'homologues (Bivalents) & Chromosomes individuels \\ \hline
            \textbf{Anaphase} & Séparation \textbf{homologues} (Centromère intact) & Séparation \textbf{chromatides} (Centromère divisé) \\ \hline
            \textbf{Brassage} & Inter \textbf{ET} Intra-chromosomique & Aucun nouveau brassage \\ \hline
            \textbf{Résultat ADN} & 2 cellules n, 2c ADN (chromo 2 chromatides) & 4 cellules n, 1c ADN (chromo 1 chromatide) \\ \hline
        \end{tabularx}
    \end{center}
\end{methode}

\begin{methode}[Expliquer le maintien de la quantité d'ADN au cours de la reproduction sexuée]
    \textbf{Objectif :} Articuler \cle{Duplication} (Mitose/Méiose), \cle{Méiose} et \cle{Fécondation} pour montrer la constance du nombre de chromosomes et de la quantité d'ADN d'une génération à l'autre.
    \begin{enumerate}
        \item \textbf{État de référence :} Cellule somatique adulte = \textbf{2n (diploïde), 2c ADN} (chromosomes à 1 chromatide en G1).
        \item \textbf{Étape 1 : Duplication (Phase S pré-méiotique).} ADN double. Cellule 2n, \textbf{4c ADN} (chromosomes à 2 chromatides). \textit{Quantité d'ADN doublée.}
        \item \textbf{Étape 2 : Méiose (I + II).} 1 division réductionnelle + 1 division équationnelle. \textbf{1 cellule 2n/4c $\rightarrow$ 4 cellules n/1c (Gamètes).} \textit{Quantité d'ADN divisée par 4 par rapport à la cellule mère méiotique (ou divisée par 2 par rapport à la cellule somatique G1).}
        \item \textbf{Étape 3 : Fécondation.} Fusion de \textbf{2 gamètes} (n/1c + n/1c) $\rightarrow$ \textbf{Zygote 2n/2c}. \textit{Restauration de la diploïdie et de la quantité d'ADN somatique.}
        \item \textbf{Bilan global :} \textbf{2n/2c (Parent) $\xrightarrow{\text{Méiose}}$ n/1c (Gamètes) $\xrightarrow{\text{Fécondation}}$ 2n/2c (Enfant).} La constance de l'espèce est assurée.
    \end{enumerate}
    \textbf{Schéma bilan (flèches) :} \fbox{Cellule Somatique 2n/2c} $\xrightarrow{\text{Phase S}}$ \fbox{2n/4c} $\xrightarrow{\text{Méiose I}}$ \fbox{n/2c} $\xrightarrow{\text{Méiose II}}$ \fbox{4 Gamètes n/1c} $\xrightarrow{\text{Fécondation (x2)}}$ \fbox{Zygote 2n/2c}.
\end{methode}

\begin{attention}[Les 5 erreurs qui coûtent des points au Bac (Terminale Littéraire - Sciences)]
    \begin{enumerate}
        \item \textbf{Confondre "Brins" et "Chromatides" / "Chromosomes" :} Dire "l'ADN se duplique en 2 brins" (faux, c'est 2 brins $\rightarrow$ 4 brins = 2 chromatides). Dire "en Anaphase I, les chromatides se séparent" (FAUX GRAVE : ce sont les homologues ; les chromatides se séparent en Anaphase II / Mitose). \textbf{Repère :} Centromère intact = 1 chromosome (2 chromatides). Centromère divisé = 2 chromosomes (1 chromatide chacun).
        \item \textbf{Inverser les sens 5' $\rightarrow$ 3' (Transcription/Traduction/Duplication) :} L'ARN Polymérase et l'ADN Polymérase synthétisent \textbf{uniquement} en ajoutant des nucléotides sur le 3'-OH du brin en croissance $\rightarrow$ Sens de synthèse \textbf{5' $\rightarrow$ 3'}. Le brin matrice est lu \textbf{3' $\rightarrow$ 5'}. L'ARNm est lu \textbf{5' $\rightarrow$ 3'} au ribosome.
        \item \textbf{Oublier la maturation de l'ARNm (Épissage) chez l'eucaryote :} Écrire que l'ARNm sort du noyau tel quel. \textbf{Obligatoire :} Retrait des introns, jonction des exons, coiffe 5', queue poly-A 3'. Sans cela, pas de traduction correcte (décalage de cadre, codons STOP précoces).
        \item \textbf{Confondre Brassage Inter- et Intra-chromosomique :}
            \begin{itemize}
                \item \textbf{Inter} = Ségrégation aléatoire des paires d'homologues (Métaphase I/Anaphase I). Mélange chromosomes père/mère.
                \item \textbf{Intra} = Crossing-over aux chiasmes (Prophase I). Échange de segments entre chromatides non-sœurs d'un même couple homologue. Mélange allèles \emph{sur un même chromosome}.
            \end{itemize}
        \item \textbf{Écrire "ADN $\rightarrow$ Protéine" directement (Transcription inverse oubliée) :} Le dogme central est \textbf{ADN $\xrightarrow{\text{Transcription}}$ ARN $\xrightarrow{\text{Traduction}}$ Protéine}. Jamais ADN direct vers protéine (sauf rétrovirus, hors programme). Ne pas citer l'ARNm, l'ARNt, le ribosome = 0 point sur la question "décrivez la biosynthèse".
    \end{enumerate}
    \textbf{Astuce finale :} Dessinez un petit schéma annoté (fourche réplication, ribosome sur ARNm, plaque métaphase I vs II) dans votre brouillon, cela structure la rédaction et évite les confusions vocabulaire.
\end{attention}

\coursec{Le maintien de la quantité d'ADN au cours de la reproduction}

\courssub{Les acides nucléiques : ADN et ARN}

\begin{definition}[Acide désoxyribonucléique (ADN)]
L'\textbf{ADN} est le support de l'information génétique chez la quasi-totalité des êtres vivants. C'est un polymère constitué de \textbf{nucléotides} (monomères). Chaque nucléotide comprend :
\begin{itemize}
    \item un sucre : le \textbf{désoxyribose} (pentose) ;
    \item un groupe \textbf{phosphate} ;
    \item une \textbf{base azotée} : \textbf{Adénine (A)}, \textbf{Thymine (T)}, \textbf{Cytosine (C)}, \textbf{Guanine (G)}.
\end{itemize}
L'ADN est une molécule \textbf{double brin} (bibrins) enroulée en \textbf{double hélice} (hélice B). Les deux brins sont \textbf{antiparallèles} (sens 5' $\to$ 3' et 3' $\to$ 5') et \textbf{complémentaires} grâce aux liaisons hydrogène entre bases : \textbf{A = T} (2 liaisons) et \textbf{C $\equiv$ G} (3 liaisons).
\end{definition}

\begin{definition}[Acide ribonucléique (ARN)]
L'\textbf{ARN} est généralement \textbf{monobrin}. Ses nucléotides portent un \textbf{ribose} (au lieu du désoxyribose) et la base \textbf{Uracile (U)} remplace la Thymine. On distingue trois types majeurs :
\begin{itemize}
    \item \textbf{ARN messager (ARNm)} : copie d'un gène, transporte l'information du noyau au cytoplasme.
    \item \textbf{ARN de transfert (ARNt)} : adaptateur qui porte un acide aminé spécifique et reconnaît un codon sur l'ARNm grâce à son \textbf{anticodon}.
    \item \textbf{ARN ribosomique (ARNr)} : composant structural et catalytique des ribosomes.
\end{itemize}
\end{definition}

\begin{propriete}[Règles de Chargaff]
Dans l'ADN double brin, quelles que soient l'espèce ou la cellule considérée :
\begin{enumerate}
    \item \textbf{A = T} et \textbf{C = G} (égalité des bases complémentaires).
    \item \textbf{A + G = T + C} (équivalence bases puriques / pyrimidiques).
    \item \textbf{(A+T) / (C+G)} est constant pour une espèce donnée (variabilité interspécifique).
\end{enumerate}
Ces règles découlent directement de la complémentarité des brins (A face à T, C face à G).
\end{propriete}

\begin{exemplebox}[Application des règles de Chargaff]
\textbf{Donnée :} ADN humain, $\%A = 30\%$.
\textbf{Déduction :} $\%T = 30\%$ (règle 1). Reste $100 - 60 = 40\%$ pour C+G. Donc $\%C = 20\%$ et $\%G = 20\%$.
\textbf{Intérêt :} Le dosage des bases permet d'identifier une espèce ou de vérifier la pureté d'un extrait d'ADN (ex: au Centre Pasteur du Cameroun pour le typage de souches bactériennes).
\end{exemplebox}

\begin{savaistu}[L'ADN au service de la justice et de la santé au Cameroun]
L'empreinte génétique (profil ADN) repose sur la variabilité des séquences non codantes (microsatellites). Elle est utilisée :
\begin{itemize}
    \item aux \textbf{Laboratoires de la Police Scientifique (Yaoundé, Douala)} pour l'identification de personnes (affaires criminelles, paternité) ;
    \item au \textbf{Centre de Recherche sur les Maladies Infectieuses (CREMER)} pour le typage des souches de \emph{Mycobacterium tuberculosis} (tuberculose) ou de \emph{Plasmodium falciparum} (paludisme) afin de suivre la résistance aux traitements.
\end{itemize}
\end{savaistu}

\courssub{La duplication de l'ADN}

\begin{definition}[Duplication (Réplication) de l'ADN]
C'est le processus par lequel une molécule d'ADN double brin donne \textbf{deux molécules filles identiques} à la molécule mère. Elle a lieu dans le \textbf{noyau} (eucaryotes) pendant la \textbf{phase S} du cycle cellulaire, avant toute division (mitose ou méiose).
\end{definition}

\begin{propriete}[Mode semi-conservatif]
Chaque molécule fille conserve \textbf{un brin matrice (parental)} et un \textbf{brin néo-synthétisé}. Ce mode assure la \textbf{fidélité} de la transmission : l'information d'un brin sert de matrice pour la synthèse exacte du brin complémentaire (règle A=T, C=G).
\end{propriete}

\begin{experience}[L'expérience de Meselson et Stahl (1958) — Preuve du mode semi-conservatif]
\textbf{Organisme :} Bactérie \emph{Escherichia coli} (division rapide, ADN unique).
\textbf{Principe :} Marquage de l'ADN par un isotope lourd de l'azote (\ce{^{15}N}, milieu "lourd") $\to$ transfert en milieu léger (\ce{^{14}N}) $\to$ suivi de la densité de l'ADN par \textbf{centrifugation isopycnique} (gradient de CsCl).
\begin{center}
\begin{tabular}{|c|c|c|}\hline
\textbf{Génération} & \textbf{Milieu} & \textbf{Bande(s) observée(s)} \\ \hline
0 (témoin) & \ce{^{15}N} & 1 bande \textbf{Lourde (L)} \\ \hline
1 & \ce{^{14}N} & 1 bande \textbf{Intermédiaire (I)} \\ \hline
2 & \ce{^{14}N} & 2 bandes : \textbf{Intermédiaire (I)} et \textbf{Légère (L)} \\ \hline
\end{tabular}
\end{center}
\textbf{Interprétation :}
\begin{itemize}
    \item Génération 1 : Une seule bande I $\to$ élimine le mode \textbf{conservatif} (qui donnerait L + L) et le mode \textbf{dispersif} (qui donnerait une bande I unique mais qui se déplacerait différemment aux générations suivantes).
    \item Génération 2 : Bandes I + L $\to$ confirme le mode \textbf{semi-conservatif} : les molécules I (un brin \ce{^{15}N}, un brin \ce{^{14}N}) se dupliquent pour donner une molécule I et une molécule L (deux brins \ce{^{14}N}).
\end{itemize}
\end{experience}

\begin{methode}[Interpréter une expérience de marquage isotopique]
\begin{enumerate}
    \item Identifier le marquage initial (lourd/léger) et le milieu de culture suivant.
    \item Prédire la position des bandes pour chaque hypothèse (conservatif, semi-conservatif, dispersif).
    \item Comparer aux résultats expérimentaux (nombre et position des bandes).
    \item Conclure sur le mode de réplication.
\end{enumerate}
\end{methode}

\begin{aretenir}[Fidélité de la duplication]
La fidélité repose sur deux piliers :
\begin{enumerate}
    \item \textbf{Complémentarité des bases} : A s'apparie strictement avec T, C avec G.
    \item \textbf{Activité de relecture (\emph{proofreading}) de l'ADN polymérase} : elle détecte et excise (exonucléase 3'$\to$5') les bases mal appariées \emph{avant} de continuer la synthèse. Taux d'erreur $\approx 10^{-9}$ à $10^{-10}$ par base.
\end{enumerate}
\end{aretenir}

\courssub{La biosynthèse des protéines}

\begin{definition}[Gène]
Un \textbf{gène} est une portion d'ADN (suite de nucléotides) qui contient l'information pour la synthèse d'une \textbf{protéine} (ou d'un ARN fonctionnel). Chez l'eucaryote, il contient des \textbf{exons} (parties codantes conservées) et des \textbf{introns} (parties non codantes éliminées).
\end{definition}

\begin{propriete}[Le code génétique]
C'est le dictionnaire qui traduit la langue des nucléotides (ARNm) en langue des acides aminés.
\begin{itemize}
    \item \textbf{Codon} : triplet de bases sur l'ARNm (ex: AUG). 64 codons possibles.
    \item \textbf{Dégénéré} : plusieurs codons pour un même acide aminé (ex: 4 codons pour l'Alanine).
    \item \textbf{Universel} (quasi) : même code du virus à l'homme $\to$ preuve d'origine commune.
    \item \textbf{Sans chevauchement, sans ponctuation} : lecture continue triplet par triplet.
    \item \textbf{Codon START} : AUG (Méthionine). \textbf{Codons STOP} : UAA, UAG, UGA.
\end{itemize}
\end{propriete}

\begin{methode}[De l'ADN à la protéine : les 3 étapes majeures]
\begin{enumerate}
    \item \textbf{Transcription (Noyau)} : L'\textbf{ARN polymérase} synthétise un \textbf{pré-ARNm} complémentaire au brin matrice (matrice 3'$\to$5', synthèse 5'$\to$3'). Remplacement T par U.
    \item \textbf{Maturation / Épissage (Noyau)} : Élimination des \textbf{introns}, jonction des \textbf{exons} $\to$ \textbf{ARNm mature}. Ajout coiffe 5' et queue poly-A 3'. Sortie par les pores nucléaires.
    \item \textbf{Traduction (Cytoplasme, Ribosome)} : Lecture de l'ARNm codon par codon. Les \textbf{ARNt} apportent les acides aminés correspondants. L'enzyme \textbf{peptidyl-transférase} (activité de l'ARNr) forme les \textbf{liaisons peptidiques}. Arrêt au codon STOP $\to$ libération de la chaîne polypeptidique.
\end{enumerate}
\end{methode}

\begin{exemplebox}[La drépanocytose : quand un codon change tout]
\textbf{Contexte camerounais :} Maladie génétique fréquente (prévalence trait drépanocytaire $\approx$ 20-25\% dans certaines zones).
\begin{itemize}
    \item \textbf{Gène :} \emph{HBB} (chromosome 11), chaîne $\beta$-globine de l'hémoglobine.
    \item \textbf{Allèle normal ($A$) :} Codon 6 = \textbf{GAG} $\to$ \textbf{Glutamate (Glu)} (acide aminé chargé, soluble).
    \item \textbf{Allèle muté ($S$) :} Mutation \textbf{ponctuelle de substitution} (A $\to$ T sur brin codant) $\to$ Codon 6 = \textbf{GTG} (ARNm \textbf{GUG}) $\to$ \textbf{Valine (Val)} (acide aminé hydrophobe).
    \item \textbf{Conséquence :} HbS (hémoglobine S) polymérise en fibres rigides en cas d'hypoxie $\to$ globules rouges en "faucille" (drépanocytes) $\to$ anémie hémolytique, crises vaso-occlusives, infections.
    \item \textbf{Avantage hétérozygote (AS) :} Résistance au paludisme grave (\emph{Plasmodium falciparum}) $\to$ maintien de l'allèle $S$ dans les zones d'endémie palustre (Sélection équilibrante).
\end{itemize}
\end{exemplebox}

\courssub{La méiose : de la diploïdie à l'haploïdie}

\begin{definition}[Méiose]
Suite de \textbf{deux divisions successives} (Méiose I et II) sans réplication d'ADN intercalaire, transformant une cellule \textbf{diploïde (2n, 2C en G1)} en \textbf{quatre cellules haploïdes (n, 1C)} : les \textbf{gamètes}. Elle a lieu dans les \textbf{gonades} (testicules : spermatogenèse ; ovaires : ovogenèse).
\end{definition}

\begin{propriete}[Méiose I : Division réductionnelle]
Sépare les \textbf{chromosomes homologues} (pères/mères).
\begin{itemize}
    \item \textbf{Prophase I :} \textbf{Appariement} des homologues (synapse) $\to$ \textbf{Crossing-over} (échange de segments entre chromatides non-sœurs des homologues) $\to$ \textbf{Brassage intrachromosomique}. Formation des chiasmas.
    \item \textbf{Métaphase I :} Alignement des paires d'homologues (bivalents) au plan équatorial. Orientation aléatoire $\to$ \textbf{Brassage interchromosomique} ($2^n$ combinaisons possibles).
    \item \textbf{Anaphase I :} Séparation des homologues (chiasmas libérés). Migration vers les pôles.
    \item \textbf{Télophase I :} Deux cellules \textbf{haploïdes (n)} mais chromosomes encore \textbf{dupliqués (2 chromatides)}. Quantité ADN = 2C.
\end{itemize}
\end{propriete}

\begin{propriete}[Méiose II : Division équationnelle]
Sépare les \textbf{chromatides sœurs} (comme en mitose).
\begin{itemize}
    \item Pas de réplication d'ADN (Interkynèse courte, pas de phase S).
    \item Anaphase II : Séparation des centromères $\to$ chromatides deviennent chromosomes filles.
    \item Résultat : \textbf{4 cellules haploïdes (n)}, chromosomes \textbf{simples (1 chromatide)}. Quantité ADN = 1C.
\end{itemize}
\end{propriete}

\begin{methode}[Analyser l'évolution de la quantité d'ADN et du nombre de chromosomes]
\begin{enumerate}
    \item \textbf{Phase S (Duplication) :} Quantité ADN double (2C $\to$ 4C). Nombre de chromosomes inchangé (2n), mais chaque chromosome = 2 chromatides.
    \item \textbf{Méiose I :} Quantité ADN divisée par 2 (4C $\to$ 2C). Nombre de chromosomes divisé par 2 (2n $\to$ n). Chromosomes à 2 chromatides.
    \item \textbf{Méiose II :} Quantité ADN divisée par 2 (2C $\to$ 1C). Nombre de chromosomes inchangé (n). Chromosomes à 1 chromatide.
    \item \textbf{Fécondation :} Fusion de 2 gamètes (n + n) $\to$ Zygote (2n, 2C). \textbf{Constante d'espèce restaurée.}
\end{enumerate}
\end{methode}

\begin{attention}[Pièges fréquents : Vocabulaire chromosomique]
\begin{itemize}
    \item \textbf{2n / n} : nombre de \textbf{chromosomes} (centromères).
    \item \textbf{2C / 1C / 4C} : quantité d'\textbf{ADN} (contenu en C = génome haploïde de référence).
    \item \textbf{Chromatide} : brin d'ADN d'un chromosome dupliqué. 1 chromosome = 1 chromatide (G1) ou 2 chromatides sœurs (G2, Méiose I).
    \item \textbf{Homologues} : même taille, même centromère, mêmes gènes (allèles éventuellement différents). Viennent l'un du père, l'autre de la mère.
    \item \textbf{Sœurs} : copies identiques issues de la réplication (sauf erreur rare).
\end{itemize}
\end{attention}

\begin{experience}[Observation de caryotypes : Trisomie 21]
\textbf{Technique :} Culture de lymphocytes (sang périphérique) ou amniocytes (liquide amniotique) $\to$ arrêt en métaphase (colchicine) $\to$ étalement $\to$ coloration (bande G) $\to$ photographie et classement par paires (taille, centromère, bandes).

\textbf{Caryotype} & \textbf{Nombre total} & \textbf{Paire 21} & \textbf{Interprétation} \\ \hline
Normal (A) & 46 (2n) & 2 chromosomes & Femme 46,XX ou Homme 46,XY \\ \hline

\textbf{Origine :} \textbf{Non-disjonction} (échec de séparation).
\begin{itemize}
    \item \textbf{Méiose I :} Homologues ne se séparent pas $\to$ gamète n+1 (2 chr 21) + gamète n-1 (0 chr 21). \textbf{Cas le plus fréquent (environ 75-80\%), surtout d'origine maternelle.}

\end{itemize}

\end{experience}

\courssub{Synthèse : Maintien de la quantité d'ADN et pérennité de l'espèce}

\begin{aretenir}[Bilan global : Duplication $\to$ Méiose $\to$ Fécondation]
\begin{center}
\begin{tabular}{|c|c|c|c|}\hline
\textbf{Étape} & \textbf{Nombre Chromosomes} & \textbf{Quantité ADN} & \textbf{Événement clé} \\ \hline
Cellule souche (G1) & 2n & 2C & Identité de l'espèce \\ \hline
\textbf{Duplication (Phase S)} & 2n & \textbf{4C} & \textbf{Fidélité} (semi-conservatif + relecture) \\ \hline
\textbf{Méiose I (Réductionnelle)} & \textbf{n} & 2C & \textbf{Diversité} (Brassage inter + intra) \\ \hline
\textbf{Méiose II (Équationnelle)} & n & \textbf{1C} & Séparation chromatides sœurs \\ \hline
\textbf{Gamètes} & n & 1C & Haploïdes, génétiquement uniques \\ \hline
\textbf{Fécondation} & \textbf{2n} & \textbf{2C} & \textbf{Restauration 2n} + Brassage fécondation \\ \hline
\end{tabular}
\end{center}
\textbf{Conclusion :} La \textbf{duplication} assure la fidélité de la copie. La \textbf{méiose} réduit le nombre de chromosomes (évite le doublement à chaque génération) et crée de la diversité (brassages). La \textbf{fécondation} restaure le nombre constant (2n) propre à l'espèce. L'espèce est ainsi pérennisée \textbf{stable dans son caryotype, diverse dans ses individus}.
\end{aretenir}

\begin{savaistu}[Diversité et adaptation au Cameroun]
La diversité génétique brassée par la méiose (crossing-over, ségrégation indépendante) et la fécondation est le carburant de l'évolution. Au Cameroun, elle explique :
\begin{itemize}
    \item La variété des réponses immunitaires face aux parasites (paludisme, onchocercose).
    \item L'adaptation de variétés locales de maïs, manioc, plantain aux sols et climats variés (zones soudano-sahéliennes vs forêt équatoriale).
    \item L'importance de conserver les \textbf{ressources génétiques} (banques de gènes à l'IRAD) pour l'amélioration variétale face au changement climatique.
\end{itemize}
\end{savaistu}

\begin{exercice}[Sujet type Bac : Synthèse — Duplication, Méiose et Fécondation]
\textbf{Consigne :} Rédigez un texte structuré (introduction, développement en deux parties, conclusion) pour répondre à la question suivante :
\textbf{« Comment la succession de la duplication de l'ADN, de la méiose et de la fécondation assure-t-elle à la fois la stabilité du nombre de chromosomes de l'espèce et la diversité génétique des individus ? »}

\textbf{Points obligatoires à traiter dans le développement :}
\begin{enumerate}
    \item \textbf{Duplication (Phase S) :} Doublement de la quantité d'ADN (2C $\to$ 4C) et du nombre de chromatides, sans changer le nombre de chromosomes (2n). Fidélité assurée par complémentarité des bases et relecture par ADN polymérase.
    \item \textbf{Méiose :} Distinction claire entre Méiose I (réductionnelle : séparation des homologues, brassage interchromosomique / \emph{crossing-over}) et Méiose II (équationnelle : séparation des chromatides sœurs). Passage de 2n (4C) $\to$ n (1C) dans les gamètes.
    \item \textbf{Fécondation :} Fusion aléatoire de deux gamètes haploïdes (n) $\to$ restauration du diploïde (2n) constant pour l'espèce. Brassage fécondation + brassage méiotique = diversité génétique.
\end{enumerate}
\end{exercice}

\begin{corrige}[Exercice 1 — QCM : Structure des acides nucléiques]
\textit{Énoncé : Se reporter à l'exercice 1 du sujet d'entraînement.}
\tcblower
\textbf{Réponse exacte : proposition 3.}

\textbf{Justification :} L'ADN est bien une molécule \cle{double brin} dont les deux brins sont \cle{antiparallèles} (l'un orienté 5'$\to$3', l'autre 3'$\to$5') et \cle{complémentaires} : l'adénine s'apparie avec la thymine (A = T, 2 liaisons hydrogène) et la cytosine avec la guanine (C $\equiv$ G, 3 liaisons hydrogène).

\textbf{Pourquoi les autres propositions sont fausses :}
\begin{enumerate}
    \item \textbf{Fausse :} elle mélange ADN et ARN. L'ADN est \emph{double brin}, contient du \emph{désoxyribose} et la base \emph{thymine} (jamais d'uracile). La description donnée (monobrin, ribose, uracile) correspond à l'ARN.
    \item \textbf{Fausse :} l'ARN est généralement \emph{monobrin} ; il ne contient pas de thymine mais de l'\emph{uracile} (A s'apparie avec U).
    \item \textbf{Fausse :} la duplication de l'ADN a lieu dans le \emph{noyau} (phase S du cycle cellulaire). Les ribosomes, situés dans le cytoplasme, sont le siège de la \emph{traduction}, pas de la réplication.
\end{enumerate}
\end{corrige}

\begin{corrige}[Exercice 2 — Vrai ou Faux : Duplication et code génétique]
\textit{Énoncé : Se reporter à l'exercice 2 du sujet d'entraînement.}
\tcblower
\begin{enumerate}
    \item \textbf{FAUX.} La réplication est \cle{semi-conservative} : chaque molécule fille conserve \textbf{un} brin parental (matrice) et possède \textbf{un} brin néo-synthétisé. Le mode « conservatif » (molécule mère intacte + molécule fille entièrement neuve) a été éliminé par l'expérience de Meselson et Stahl.
    \item \textbf{VRAI.} C'est la \cle{transcription} : l'ARN polymérase synthétise dans le noyau un ARNm complémentaire du brin matrice d'ADN ; après maturation (épissage des introns), l'ARNm mature sort par les pores nucléaires et est traduit sur les ribosomes du cytoplasme.
    \item \textbf{FAUX.} Un codon est un triplet de nucléotides porté par l'\textbf{ARNm} (et non par l'ADN). C'est sur l'ARNm que le ribosome lit le message, codon par codon. (Sur l'ADN, on parle de « triplet » du brin codant.)
    \item \textbf{FAUX.} La méiose I est la division \cle{réductionnelle} : elle sépare les \emph{chromosomes homologues} et fait passer la cellule de 2n à n. C'est la méiose II qui est \cle{équationnelle} : elle sépare les \emph{chromatides sœurs} (comme une mitose).
\end{enumerate}
\end{corrige}

\begin{corrige}[Exercice 3 — Définitions et pourcentages (Données de Chargaff)]
\textit{Énoncé : Se reporter à l'exercice 3 du sujet d'entraînement.}
\tcblower
\textbf{1. Déduction des pourcentages.}

D'après la première règle de Chargaff (complémentarité des bases) : $\%A = \%T$ et $\%C = \%G$.
\begin{align*}
\%T &= \%A = 30\,\% \\
\%C + \%G &= 100\,\% - (\%A + \%T) = 100 - 60 = 40\,\% \\
\%C &= \%G = \frac{40\,\%}{2} = 20\,\%
\end{align*}
\textbf{Bilan :} A = 30\,\%, T = 30\,\%, C = 20\,\%, G = 20\,\%. Vérification : $30+30+20+20 = 100\,\%$. 

\textbf{2. Règle générale illustrée.}

Ces résultats illustrent les \cle{règles de Chargaff} : dans tout ADN double brin, $\%A = \%T$ et $\%C = \%G$ (et donc $\%A + \%G = \%T + \%C = 50\,\%$), conséquence directe de la complémentarité des deux brins. Le rapport (A+T)/(C+G), ici égal à $60/40 = 1,5$, est caractéristique de l'espèce.

\textbf{3. Définitions.}
\begin{itemize}
    \item \textbf{Gène :} portion d'ADN portant l'information nécessaire à la synthèse d'une protéine (ou d'un ARN fonctionnel).
    \item \textbf{Allèle :} version d'un même gène, occupant le même locus sur les chromosomes homologues (ex. allèles $A$ et $S$ du gène \emph{HBB}).
    \item \textbf{Caryotype :} ensemble des chromosomes d'une cellule, photographiés en métaphase, classés par paires d'homologues selon leur taille et la position du centromère (ex. 46,XX).
\end{itemize}
\end{corrige}

\begin{corrige}[Exercice 4 — Ordre chronologique : De l'ADN à la protéine]
\textit{Énoncé : Se reporter à l'exercice 4 du sujet d'entraînement.}
\tcblower
\textbf{Ordre correct :}
\begin{enumerate}
    \item \textbf{Transcription dans le noyau} (synthèse du pré-ARNm par l'ARN polymérase).
    \item \textbf{Maturation de l'ARNm (épissage) et sortie vers le cytoplasme} (élimination des introns, jonction des exons).
    \item \textbf{Reconnaissance du codon START (AUG) par l'ARNt initiateur} (initiation de la traduction).
    \item \textbf{Fixation de l'ARNt porteur d'acide aminé sur le codon de l'ARNm} (élongation : appariement codon/anticodon).
    \item \textbf{Allongement de la chaîne polypeptidique} (formation des liaisons peptidiques).
    \item \textbf{Traduction au niveau du ribosome (phase globale)} se termine au codon STOP : libération de la protéine.
\end{enumerate}
\textbf{Point de vigilance :} la transcription et la maturation ont lieu dans le \emph{noyau} ; l'initiation, l'élongation et la terminaison ont lieu dans le \emph{cytoplasme}, sur le ribosome. Le codon STOP ne code pour aucun acide aminé : il provoque la libération de la chaîne.
\end{corrige}

\begin{corrige}[Exercice 5 — Exploitation de l'expérience de Meselson et Stahl]
\textit{Énoncé : Se reporter à l'exercice 5 du sujet d'entraînement.}
\tcblower
\textbf{1. Les trois hypothèses historiques :}
\begin{itemize}
    \item \textbf{Conservative :} la molécule mère reste intacte ; une molécule fille entièrement nouvelle est synthétisée à côté.
    \item \textbf{Semi-conservative :} chaque molécule fille reçoit un brin parental et un brin néo-synthétisé.
    \item \textbf{Dispersive :} chaque brin fille est une mosaïque de segments parentaux et de segments neufs alternés.
\end{itemize}

\textbf{2. Élimination à partir des générations 0 et 1.}

Génération 0 : une bande lourde (ADN 100\,\% \ce{^{15}N}). Après une division en milieu \ce{^{14}N}, on observe \textbf{une seule bande intermédiaire}.
\begin{itemize}
    \item L'hypothèse \textbf{conservative} est \textbf{éliminée} : elle prévoyait à la génération 1 \emph{deux} bandes — une bande lourde (molécule mère conservée) et une bande légère (molécule fille entièrement \ce{^{14}N}). Or on n'observe qu'une bande intermédiaire.
    \item Les hypothèses semi-conservative et dispersive restent possibles à ce stade (toutes deux prévoient une bande unique de densité intermédiaire).
\end{itemize}

\textbf{3. Confirmation à la génération 2.}

On observe \textbf{deux bandes} : une intermédiaire (I) et une légère (L).
\begin{itemize}
    \item \textbf{Bande I :} molécules hybrides composées d'\textbf{un brin parental \ce{^{15}N}} et d'\textbf{un brin néo-synthétisé \ce{^{14}N}}.
    \item \textbf{Bande L :} molécules composées de \textbf{deux brins \ce{^{14}N}} (issues de la duplication des molécules hybrides : le brin \ce{^{14}N} de la molécule I a servi de matrice à un nouveau brin \ce{^{14}N}).
\end{itemize}
L'hypothèse \textbf{dispersive} est éliminée : elle prévoyait une bande unique dont la densité diminuerait progressivement à chaque génération (dilution du \ce{^{15}N} dans tous les brins), jamais deux bandes distinctes dont une légère.

\textbf{Conclusion :} seule l'hypothèse \cle{semi-conservative} rend compte de l'ensemble des résultats : chaque molécule fille conserve un brin parental, ce qui garantit la fidélité de la transmission de l'information génétique.
\end{corrige}

\begin{corrige}[Exercice 6 — Traduction guidée et mutation]
\textit{Énoncé : Se reporter à l'exercice 6 du sujet d'entraînement.}
\tcblower
\textbf{1. Séquence de l'ARNm.}

Le brin codant (non matrice) a la même séquence que l'ARNm, T étant remplacé par U :
\[ \text{ARNm : 5' -- A U G \quad G C U \quad U U A \quad C G G \quad U A A -- 3'} \]

\textbf{2. Séquence protéique.}

Lecture codon par codon à partir du codon START :
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}\hline
Codon & AUG & GCU & UUA & CGG & UAA \\ \hline
Acide aminé & Met & Ala & Leu & Arg & STOP \\ \hline
\end{tabular}
\end{center}
\textbf{Protéine : Met -- Ala -- Leu -- Arg} (la traduction s'arrête au codon UAA).

\textbf{3. Mutation de la 6\textsuperscript{e} base (T $\to$ C).}

Le brin codant devient : 5' -- A T G \; G C \textbf{C} \; T T A \; C G G \; T A A -- 3'.

Nouvel ARNm : 5' -- A U G \; G C \textbf{C} \; U U A \; C G G \; U A A -- 3'.

Le codon GCU devient \textbf{GCC}. Or, d'après la table fournie, GCU et GCC codent \emph{tous deux} pour l'\textbf{Alanine (Ala)}.

\textbf{Nouvelle chaîne peptidique : Met -- Ala -- Leu -- Arg} (identique à l'originale).

\textbf{Type de mutation :} c'est une mutation \cle{synonyme} (ou silencieuse) : la substitution change un codon mais pas l'acide aminé incorporé, grâce à la \textbf{dégénérescence du code génétique} (plusieurs codons pour un même acide aminé). Elle n'a donc aucune conséquence phénotypique.

\textbf{Point de vigilance :} ne pas confondre avec une mutation \emph{faux-sens} (changement d'acide aminé, ex. Glu $\to$ Val dans la drépanocytose) ni \emph{non-sens} (apparition d'un codon STOP prématuré).
\end{corrige}

\begin{corrige}[Exercice 7 — Analyse de courbe : Quantité d'ADN et méiose]
\textit{Énoncé : Se reporter à l'exercice 7 du sujet d'entraînement.}
\tcblower
\textbf{1. Identification des phases.}
\begin{itemize}
    \item \textbf{Duplication (phase S) :} passage de 2C à 4C entre les points $G_1$/S et $G_2$ (montée de la courbe de 2 à 4 UA).
    \item \textbf{Méiose I :} chute de 4C à 2C (première division).
    \item \textbf{Méiose II :} chute de 2C à 1C (deuxième division) $\to$ gamètes à 1C.
\end{itemize}

\textbf{2. Chromosomes et chromatides à chaque étape.}
\begin{center}
\begin{tabular}{|c|c|c|c|}\hline
\textbf{Étape} & \textbf{Chromosomes} & \textbf{Chromatides/chromosome} & \textbf{Quantité d'ADN} \\ \hline
$G_1$ & 2n & 1 chromatide (2n/1c) & 2C \\ \hline
$G_2$ (après phase S) & 2n & 2 chromatides (2n/2c) & 4C \\ \hline
Après Méiose I & n & 2 chromatides (n/2c) & 2C \\ \hline
Après Méiose II & n & 1 chromatide (n/1c) & 1C \\ \hline
\end{tabular}
\end{center}

\textbf{3. Division réductionnelle et division équationnelle.}
\begin{itemize}
    \item La \textbf{Méiose I} est \cle{réductionnelle} : elle \textbf{divise par deux le nombre de chromosomes} (2n $\to$ n) par séparation des chromosomes homologues. La quantité d'ADN passe de 4C à 2C.
    \item La \textbf{Méiose II} est \cle{équationnelle} : le \textbf{nombre de chromosomes ne change pas} (n $\to$ n) ; seules les chromatides sœurs se séparent, ce qui ramène la quantité d'ADN de 2C à 1C sans réduire le nombre de centromères.
\end{itemize}
\textbf{Justification par la courbe :} la première chute (4C $\to$ 2C) s'accompagne d'une réduction du nombre de chromosomes ; la seconde (2C $\to$ 1C) ne fait que séparer les chromatides, à nombre de chromosomes constant.
\end{corrige}

\begin{corrige}[Exercice 8 — Étude de caryotypes : Trisomie 21]
\textit{Énoncé : Se reporter à l'exercice 8 du sujet d'entraînement.}
\tcblower
\textbf{1. Anomalie et conséquence.}

Le caryotype B présente une \cle{trisomie 21} : trois exemplaires du chromosome 21 au lieu de deux (formule : 47,XX,+21 ou 47,XY,+21). Conséquence phénotypique : le \textbf{syndrome de Down} (déficience intellectuelle, traits faciaux caractéristiques, cardiopathies fréquentes).

\textbf{2. Hypothèse mécanistique : la non-disjonction.}

La trisomie résulte d'une \textbf{non-disjonction} du chromosome 21 lors de la gamétogenèse (le plus souvent maternelle) :
\begin{itemize}
    \item \textbf{Non-disjonction en Méiose I :} les deux chromosomes 21 \emph{homologues} ne se séparent pas et migrent vers le même pôle $\to$ un gamète à \textbf{n+1} (deux chromosomes 21, chacun à 2 chromatides) et un gamète à \textbf{n$-$1} (sans chromosome 21). C'est le cas le plus fréquent.
    \item \textbf{Non-disjonction en Méiose II :} les \emph{chromatides sœurs} d'un chromosome 21 ne se séparent pas $\to$ un gamète n+1, un gamète n$-$1 et deux gamètes normaux n.
\end{itemize}
Dans les deux cas, la fécondation d'un gamète n+1 (24 chromosomes) par un gamète normal n (23 chromosomes) donne un zygote à \textbf{47 chromosomes}, trisomique 21.

\textbf{3. Rôle de l'âge maternel.}

Chez la femme, l'ovogenèse débute \emph{avant la naissance} : les ovocytes entrent en méiose puis restent \textbf{bloqués en prophase I} (stade dictyotène) pendant des dizaines d'années, jusqu'à l'ovulation. Avec l'âge :
\begin{itemize}
    \item la \textbf{cohésion} entre chromatides sœurs et les \textbf{chiasmas} maintenant les homologues appariés s'affaiblissent ;
    \item le \textbf{fuseau} méiotique devient moins fiable.
\end{itemize}
Le risque de non-disjonction augmente donc avec l'âge maternel, d'où une fréquence accrue de trisomie 21 chez les enfants nés de mères de plus de 38 ans (d'où la proposition d'un dépistage prénatal dans les centres de santé de référence).
\end{corrige}

\begin{corrige}[Exercice 9 — Sujet type Bac : Fidélité de la transmission et constance du nombre de chromosomes]
\textit{Énoncé : Se reporter à l'exercice 9 du sujet d'entraînement.}
\tcblower
\textbf{Corrigé rédigé (plan attendu).}

\textbf{Introduction.} La pérennité d'une espèce exige que l'information génétique soit transmise sans erreur (fidélité) et que le nombre de chromosomes reste constant de génération en génération. Deux faits expérimentaux éclairent ces exigences.

\textbf{1. Fait expérimental 1 : l'expérience de Meselson et Stahl démontre la réplication semi-conservative, garante de la fidélité.}

Des bactéries \emph{E. coli} cultivées en milieu \ce{^{15}N} (azote lourd) possèdent un ADN lourd (bande L, génération 0). Transférées en milieu \ce{^{14}N} :
\begin{itemize}
    \item à la génération 1, une \textbf{bande unique intermédiaire} apparaît : chaque molécule fille possède un brin parental \ce{^{15}N} et un brin neuf \ce{^{14}N} ;
    \item à la génération 2, on observe \textbf{deux bandes} (intermédiaire + légère), ce qui élimine les modes conservatif et dispersif.
\end{itemize}
La réplication est donc \textbf{semi-conservative} : le brin parental conservé sert de matrice, et la complémentarité des bases (A=T, C$\equiv$G) impose la séquence du brin néo-synthétisé. L'information est recopiée fidèlement. De plus (Document 3), l'ADN polymérase possède une \textbf{activité de relecture} (exonucléase 3'$\to$5') qui excise les bases mal appariées : le taux d'erreur tombe à environ $10^{-9}$ par base.

\textbf{2. Fait expérimental 2 : la méiose maintient constant le nombre de chromosomes.}

D'après le Document 2 (espèce 2n = 4) :
\begin{itemize}
    \item en $G_2$, la cellule est 2n = 4, 4C (chromosomes à 2 chromatides) ;
    \item la \textbf{Méiose I (réductionnelle)} sépare les chromosomes \emph{homologues} : les cellules filles passent de \textbf{2n à n} (n = 2, 2C) ;
    \item la \textbf{Méiose II (équationnelle)} sépare les \emph{chromatides sœurs} : quatre gamètes \textbf{n = 2, 1C} sont produits.
\end{itemize}
Sans méiose, le nombre de chromosomes doublerait à chaque génération. La \textbf{fécondation} (fusion de deux gamètes n) restaure le nombre \textbf{2n} caractéristique de l'espèce : $n + n = 2n$.

\textbf{Conclusion.} La duplication semi-conservative (avec relecture) assure la fidélité de la copie ; le couple méiose/fécondation assure la constance du caryotype. L'espèce est ainsi transmise de façon stable et fidèle.

\begin{corrige}[Exercice 10 — Sujet type Bac : De l'information génétique au phénotype (drépanocytose)]
\textit{Énoncé : Se reporter à l'exercice 10 du sujet d'entraînement.}
\tcblower
\textbf{1. Niveau moléculaire.}

La mutation est une \textbf{substitution ponctuelle} : une seule base est remplacée (A $\to$ T sur le brin codant), ce qui transforme le codon GAG en GUG sur l'ARNm. D'après le Document 1, le \textbf{glutamate (Glu)} en position 6 de la chaîne $\beta$-globine est remplacé par une \textbf{valine (Val)} : la \emph{structure primaire} de la protéine est modifiée d'un seul acide aminé (mutation faux-sens).

Conséquence : le glutamate est un acide aminé hydrophile (chargé), la valine est \textbf{hydrophobe}. En condition d'hypoxie (manque d'oxygène), les molécules d'hémoglobine S s'agrègent par leurs valines exposées et \textbf{polymérisent en fibres rigides} : les globules rouges se déforment en faucilles, se détruisent prématurément (anémie) et obstruent les vaisseaux (crises vaso-occlusives) $\to$ pathologie grave.

\textbf{2. Niveau transmission (analyse du pedigree).}
\begin{itemize}
    \item \textbf{Transmission autosomique :} des individus des \emph{deux sexes} sont atteints (II-1 femme, II-4 et III-1 hommes, III-4 femme) ; le gène \emph{HBB} est porté par le chromosome 11 (autosome), pas par un chromosome sexuel.
    \item \textbf{Allèle $S$ récessif :} les parents I-1 et I-2 sont \emph{sains} mais ont des descendants atteints (II-1, II-4) : ils sont donc porteurs sains (hétérozygotes AS). De même, les couples II-1 $\times$ II-2 et II-3 $\times$ II-4, comprenant des individus sains, produisent des enfants atteints (III-1, III-4). Un caractère qui « saute une génération » chez des parents sains est récessif : seuls les homozygotes \textbf{SS} sont atteints ; les hétérozygotes \textbf{AS} sont sains (porteurs).
\end{itemize}

\textbf{3. Niveau population : le paradoxe de l'allèle $S$.}

À l'état homozygote SS, l'allèle est létal ou très handicapant ; il devrait donc être éliminé par la sélection naturelle. Pourtant, dans les zones d'endémie palustre (comme de nombreuses régions du Cameroun), les \textbf{hétérozygotes AS} sont \textbf{plus résistants au paludisme grave} à \emph{Plasmodium falciparum} : le parasite se développe mal dans leurs globules rouges. Les AS survivent mieux que les AA en zone impaludée et se reproduisent davantage, ce qui \textbf{maintient l'allèle $S$ à fréquence élevée} dans la population. Ce phénomène s'appelle l'\textbf{avantage sélectif de l'hétérozygote} (ou \textbf{polymorphisme équilibré / sélection équilibrante}).

Explication de la polymérisation de l'HbS (Val hydrophobe) & 4 \\ \hline
Mode de transmission justifié par des individus du pedigree & 6 \\ \hline
Avantage de l'hétérozygote AS face au paludisme + nom du phénomène & 4 \\ \hline

\end{tabular}

\textbf{Points de vigilance :}
\begin{itemize}
    \item Justifier le caractère récessif en citant explicitement des parents sains avec enfant atteint (I-1 $\times$ I-2 $\to$ II-1).
    \item Ne pas écrire que l'allèle $S$ « protège » : c'est le \emph{génotype AS} qui confère la résistance ; les SS sont gravement malades.
    \item Distinguer clairement mutation (niveau ADN), protéine (niveau moléculaire) et phénotype (niveau individu).
\end{itemize}
\end{corrige}

\begin{corrige}[Exercice 11 — Sujet type Bac : Synthèse — Duplication, Méiose et Fécondation]
\textit{Énoncé : Se reporter à l'exercice 11 du sujet d'entraînement.}
\tcblower
\textbf{Exemple de texte structuré attendu.}

\textbf{Introduction.} Chez les organismes à reproduction sexuée, chaque génération naît de la fusion de deux gamètes. Comment le nombre de chromosomes, caractéristique de l'espèce (2n = 46 chez l'Homme), peut-il rester constant alors que les individus sont tous génétiquement différents ? La réponse tient à l'enchaînement de trois événements : la duplication de l'ADN, la méiose et la fécondation.

\textbf{Développement — 1. La duplication prépare la division en copiant fidèlement l'information.}

Avant toute division, en \textbf{phase S} du cycle cellulaire, la quantité d'ADN double : la cellule passe de \textbf{2C à 4C}. Chaque chromosome, simple, devient un chromosome à \textbf{deux chromatides sœurs}, mais le nombre de chromosomes reste \textbf{2n}. Cette réplication est \textbf{semi-conservative} : chaque brin parental sert de matrice, et la complémentarité des bases (A=T, C$\equiv$G), complétée par l'activité de relecture de l'ADN polymérase, garantit une copie quasi parfaite. La duplication assure donc la \textbf{fidélité} sans laquelle aucune transmission stable n'est possible.

\textbf{2. La méiose réduit le nombre de chromosomes et brasse les allèles.}

La méiose comporte deux divisions sans réplication intermédiaire :
\begin{itemize}
    \item la \textbf{Méiose I (réductionnelle)} sépare les chromosomes \emph{homologues} : la cellule passe de \textbf{2n (4C) à n (2C)}. En prophase I, les \emph{crossing-over} échangent des segments entre homologues (\textbf{brassage intrachromosomique}) ; en métaphase I, l'orientation aléatoire des paires réalise le \textbf{brassage interchromosomique} ($2^n$ combinaisons).
    \item la \textbf{Méiose II (équationnelle)} sépare les \emph{chromatides sœurs} : on obtient \textbf{quatre gamètes haploïdes n (1C)}, tous génétiquement différents.
\end{itemize}
La méiose évite ainsi le doublement du nombre de chromosomes à chaque génération tout en créant de la \textbf{diversité}.

\textbf{3. La fécondation restaure le nombre diploïde et amplifie la diversité.}

La fusion \emph{aléatoire} de deux gamètes haploïdes ($n + n$) restaure le zygote \textbf{diploïde 2n (2C)}, constant pour l'espèce. Chaque gamète étant unique, le nombre de zygotes possibles est considérable : la fécondation ajoute un \textbf{troisième brassage} aux deux brassages méiotiques.

\textbf{Conclusion.} La duplication garantit la fidélité de la copie, la méiose réduit le nombre de chromosomes de moitié en brassant les allèles, et la fécondation restaure le 2n de l'espèce. L'espèce se perpétue ainsi \textbf{stable dans son caryotype, infiniment diverse dans ses individus} — diversité qui est le moteur de l'adaptation et de l'évolution.

\begin{itemize}
    \item Citer les valeurs 2n/n et 2C/4C/1C à chaque étape : c'est la preuve chiffrée de la stabilité.
    \item Nommer explicitement les deux brassages méiotiques (interchromosomique et intrachromosomique par crossing-over) : ce sont eux qui fondent la diversité.
    \item Ne pas écrire que la méiose « divise l'ADN par quatre » en une seule étape : la réduction se fait en \emph{deux} divisions successives.
\end{itemize}
\end{corrige}

\begin{bilan}

\textbf{Définitions clés à connaître par cœur}
\begin{itemize}
  \item \cle{ADN (Acide DésoxyriboNucléique)} : Molécule support de l'information héréditaire, double hélice antiparallèle, bases azotées \cle{A, T, G, C}, complémentarité A=T et G$\equiv$C.
  \item \cle{ARN (Acide RiboNucléique)} : Monobrin, base \cle{U} (Uracile) remplace T. Types : ARNm (messager), ARNt (transfert), ARNr (ribosomal).
  \item \cle{Gène} : Segment d'ADN codant pour une protéine (ou un ARNr/ARNt). \cle{Allèle} : Version d'un gène.
  \item \cle{Duplication (Réplication)} : Copie fidèle de l'ADN avant division cellulaire (mode semi-conservatif : 1 brin parent + 1 brin neuf).
  \item \cle{Transcription} : Synthèse d'ARNm sur un brin d'ADN (dans le noyau). \cle{Traduction} : Synthèse d'une protéine sur un ARNm au ribosome (cytoplasme).
  \item \cle{Méiose} : Division cellulaire en deux temps (I et II) transformant une cellule diploïde (2n) en 4 cellules haploïdes (n) : les gamètes.
  \item \cle{Fécondation} : Fusion de deux gamètes (n + n) restaurant la diploïdie (2n) : le zygote.
\end{itemize}

\textbf{Repères essentiels (à mémoriser)}
\begin{center}
\begin{tabularx}{\linewidth}{|l|X|X|}\hline
\rowcolor{popblueL}
\textbf{Processus} & \textbf{Lieu} & \textbf{Acteurs / Produits principaux} \\ \hline

Transcription & Noyau & ARN polymérase, ADN $\to$ ARNm (maturation : épissage) \\ \hline

\textbf{Méthodes express}
\begin{methode}[Construire une maquette ADN simple]
\begin{enumerate}
  \item Fil de fer ou pâte à modeler pour le squelette (sucre-phosphate).
  \item 4 couleurs de perles/boules pour les bases : A (popred), T (popblue), G (popgreen), C (poporange).
  \item Associer A-T (2 liens H) et G-C (3 liens H) sur deux brins antiparallèles.
  \item Torsader pour former la double hélice.
\end{enumerate}
\end{methode}

\begin{methode}[Identifier les étapes de la biosynthèse des protéines]
\begin{enumerate}
  \item \textbf{1. Transcription :} Repérer le gène sur l'ADN $\to$ synthèse ARNm pré-messager $\to$ maturation (introns retirés) $\to$ ARNm mature sort du noyau.
  \item \textbf{2. Traduction :} ARNm fixé sur ribosome $\to$ lecture codons (3 bases) $\to$ ARNt apportent acides aminés correspondants $\to$ chaîne polypeptidique s'allonge $\to$ STOP $\to$ protéine libérée.
\end{enumerate}
\end{methode}

\begin{methode}[Différencier Méiose I et Méiose II]
\begin{enumerate}
  \item \textbf{Méiose I (Réductionnelle) :} Prophase I (appariement homologues, \cle{crossing-over}), Métaphase I (paires homologues à l'équateur), Anaphase I (séparation homologues), Télophase I. \textbf{Résultat :} 2 cellules n (chaque chr = 2 chromatides).
  \item \textbf{Méiose II (Équationnelle) :} Similaire à mitose mais sans réplication préalable. Séparation des chromatides sœurs. \textbf{Résultat :} 4 cellules n (chaque chr = 1 chromatide).
\end{enumerate}
\end{methode}
