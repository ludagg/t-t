\begin{exercice}[Rappels de cours et définitions]
\begin{enumerate}
    \item Donnez la définition d'une pile électrochimique.
    \item Écrire la notation conventionnelle de la demi-pile à hydrogène standard (ESH) et rappeler sa potentialité standard.
    \item Qu'est-ce que la corrosion d'un métal ? Citer deux manifestations courantes.
    \item Citer trois facteurs favorisant la corrosion.
\end{enumerate}
\end{exercice}

\begin{exercice}[QCM : Une seule réponse correcte par question.]
\begin{enumerate}
    \item La représentation conventionnelle correcte de la pile Daniell (\ce{Zn} anode, \ce{Cu} cathode) est :
    \begin{enumerate}
        \item \ce{Zn | Zn^{2+} || Cu^{2+} | Cu}
        \item \ce{Cu | Cu^{2+} || Zn^{2+} | Zn}
        \item \ce{Zn^{2+} | Zn || Cu | Cu^{2+}}
    \end{enumerate}
    \item L'équation-bilan du fonctionnement de la pile Daniell est :
    \begin{enumerate}
        \item \ce{Zn + Cu^{2+} -> Zn^{2+} + Cu}
        \item \ce{Zn^{2+} + Cu -> Zn + Cu^{2+}}
        \item \ce{Zn + Cu -> Zn^{2+} + Cu^{2+}}
    \end{enumerate}
    \item La protection cathodique par anode sacrificielle utilise un métal :
    \begin{enumerate}
        \item Plus noble (potentiel plus haut) que le métal à protéger.
        \item Moins noble (potentiel plus bas) que le métal à protéger.
        \item De même potentiel que le métal à protéger.
    \end{enumerate}
    \item Dans une micropile de corrosion du fer en milieu aéracé, la réaction cathodique principale est :
    \begin{enumerate}
        \item \ce{Fe -> Fe^{2+} + 2e-}
        \item \ce{O2 + 2H2O + 4e- -> 4HO-}
        \item \ce{2H+ + 2e- -> H2}
    \end{enumerate}
\end{enumerate}
\end{exercice}

    \begin{center}

    \end{tabular}
    \end{center}

\begin{exercice}[Prévision de réactions et règle du gamma]
On dispose des couples redox suivants avec leurs potentiels standards :
$\ce{Ag+/Ag}$ : $\SI{+0,80}{V}$ ; $\ce{Fe^{3+}/Fe^{2+}}$ : $\SI{+0,77}{V}$ ; $\ce{Cu^{2+}/Cu}$ : $\SI{+0,34}{V}$ ; $\ce{Fe^{2+}/Fe}$ : $\SI{-0,44}{V}$ ; $\ce{Zn^{2+}/Zn}$ : $\SI{-0,76}{V}$.
\begin{enumerate}
    \item Classer les oxydants par ordre de force croissante.
    \item Préciser, en appliquant la règle du gamma, si une réaction spontanée a lieu lors du mélange des solutions suivantes. Si oui, écrire l'équation-bilan.
    \begin{enumerate}
        \item Solution de \ce{AgNO3} + métal \ce{Cu}.
        \item Solution de \ce{FeSO4} + métal \ce{Zn}.
        \item Solution de \ce{Fe2(SO4)3} + métal \ce{Cu}.
    \end{enumerate}
\end{enumerate}
\end{exercice}
