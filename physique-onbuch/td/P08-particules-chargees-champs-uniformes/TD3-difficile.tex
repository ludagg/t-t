\begin{exercice}[Spectrographe de masse à secteur magnétique et filtre de Wien]
On souhaite séparer les isotopes \ce{^{35}Cl} et \ce{^{37}Cl} (chargés \SI{+1}{e}) à l'aide d'un spectrographe de masse de type Bainbridge. Les ions sont produits au repos, accélérés par une différence de potentiel $U$, traversent un filtre de Wien (champ électrique $\vec{E}_1$ et champ magnétique $\vec{B}_1$ perpendiculaires), puis entrent dans une région de champ magnétique uniforme $\vec{B}_2$ seul (secteur à $180^\circ$) où ils sont détectés sur une plaque photographique.
Données : $m(\ce{^{35}Cl}) \approx 35\,u$, $m(\ce{^{37}Cl}) \approx 37\,u$, $1\,u = \SI{1.66e-27}{kg}$, $e = \SI{1.60e-19}{C}$.

\begin{enumerate}
    \item \textbf{Accélération et sélection de vitesse.} 
    \begin{enumerate}
        \item Établir l'expression de la vitesse $v$ des ions à la sortie de la zone d'accélération en fonction de $U$, $q$, $m$.
        \item Dans le filtre de Wien, les champs sont réglés pour ne laisser passer que les ions de vitesse $v$. Déterminer la relation entre $E_1$, $B_1$ et $v$. Justifier pourquoi cette sélection est indépendante de la masse et de la charge des ions.
    \end{enumerate}
    \item \textbf{Séparation dans le secteur magnétique.} 
    \begin{enumerate}
        \item Montrer que le rayon de courbure dans la zone $\vec{B}_2$ s'écrit $R = \frac{1}{B_2}\sqrt{\frac{2mU}{q}}$.
        \item Calculer l'écart $\Delta x$ entre les traces des deux isotopes sur la plaque détectrice (diamètre du demi-cercle) pour $U = \SI{2.00}{kV}$ et $B_2 = \SI{0.100}{T}$.
        \item Discuter de l'influence de la résolution du détecteur (largeur de fente) sur la capacité à distinguer les deux isotopes. Proposer une modification unique du dispositif (paramètre unique) pour doubler $\Delta x$ sans changer $U$ ni $B_2$.
    \end{enumerate}
    \item \textbf{Analyse critique : résolution massique.} 
    On définit le pouvoir de résolution massique $R_m = \frac{m}{\Delta m}$ où $\Delta m$ est la plus petite différence de masse séparable. En supposant que la limite est fixée par l'épaisseur $\delta$ de la trace sur la plaque ($\Delta x \ge \delta$), établir l'expression de $R_m$ en fonction de $m$, $\delta$, $U$, $B_2$, $q$. Comment évolue $R_m$ si l'on augmente $U$ ?
\end{enumerate}
\end{exercice}

\begin{exercice}[Cyclotron : limites relativistes et synchronisme]
Un cyclotron classique accélère des protons ($m_p = \SI{1.67e-27}{kg}$, $q = +e$) à l'aide d'un champ magnétique uniforme $\vec{B} = B\,\vec{u}_z$ et d'une tension alternative de fréquence $f$ appliquée entre les deux « dees ». On néglige l'espace entre les dees.
Données : $c = \SI{3.00e8}{m/s}$.

\begin{enumerate}
    \item \textbf{Fonctionnement non relativiste.} 
    \begin{enumerate}
        \item Rappeler la condition de synchronisme (égalité des périodes) et en déduire l'expression de la fréquence $f$ en fonction de $B$, $q$, $m_p$. Calculer $f$ pour $B = \SI{1.50}{T}$.
        \item Montrer que l'énergie cinétique acquise après $n$ passages dans l'interstice (tension de crête $U_0$) est $E_c = n q U_0$. En déduire le rayon $R_n$ de l'orbite au $n$-ième tour.
    \end{enumerate}
    \item \textbf{Seuil relativiste.} 
    \begin{enumerate}
        \item À partir de quelle énergie cinétique $E_c^{\text{rel}}$ la correction relativiste sur la masse ($m = \gamma m_0$) dépasse \SI{1}{\%} ? (On utilisera $\gamma = 1/\sqrt{1-v^2/c^2} \approx 1 + v^2/(2c^2)$ pour $v \ll c$).
        \item Pour $U_0 = \SI{50}{kV}$, déterminer le nombre de tours $n_{\text{rel}}$ correspondant à ce seuil. Quel est le rayon $R_{\text{rel}}$ à ce moment ?
    \end{enumerate}
    \item \textbf{Perte de synchronisme.} 
    \begin{enumerate}
        \item Exprimer la période de révolution $T$ en fonction de l'énergie totale $E = E_c + m_0c^2$ et de $B$, $q$.
        \item Si la fréquence $f$ du générateur reste fixe à sa valeur non relativiste, montrer que le déphasage $\Delta \phi$ accumulé entre l'arrivée du proton et le pic de tension après $N$ tours supplémentaires au-delà de $n_{\text{rel}}$ vaut approximativement $\Delta \phi \approx \pi \frac{N^2 q U_0}{m_0 c^2}$ (en supposant $E_c \ll m_0c^2$ mais correction sensible).
        \item Conclure sur la limitation fondamentale du cyclotron classique pour les protons et nommer la technologie qui permet de lever cette limitation.
    \end{enumerate}
\end{enumerate}
\end{exercice}

\begin{exercice}[Déviation électrique dans un oscilloscope : géométrie et sensibilité]
Dans un tube cathodique (oscilloscope), des électrons ($m_e$, $-e$) sortent d'un canon à une vitesse $v_0$ parallèle à l'axe $(Ox)$. Ils traversent deux plaques de déviation horizontales de longueur $L$, écartées de $d$, soumises à une tension $V_d(t) = V_0 \sin(\omega t)$ (champ $\vec{E} = E_y\,\vec{u}_y$). L'écran est à une distance $D$ de la sortie des plaques ($D \gg L$). On note $y_e$ l'ordonnée à la sortie des plaques et $y_{ec}$ l'ordonnée d'impact sur l'écran.

\begin{enumerate}
    \item \textbf{Trajectoire dans les plaques.} 
    \begin{enumerate}
        \item Établir les équations horaires $x(t)$ et $y(t)$ pour $0 \le t \le T_L = L/v_0$ (entrée à $t=0$, $x=0$, $y=0$). Montrer que la trajectoire est une parabole.
        \item Déterminer l'angle de sortie $\alpha = \tan^{-1}(v_y/v_x)$ et l'ordonnée $y_e$ à la sortie des plaques en fonction de $V_0$, $\omega$, $L$, $v_0$, $d$, $e$, $m_e$.
    \end{enumerate}
    \item \textbf{Position sur l'écran : régime statique et dynamique.} 
    \begin{enumerate}
        \item Exprimer $y_{ec}$ en fonction de $y_e$, $\alpha$, $D$. Simplifier pour le cas statique ($\omega = 0$, tension continue $V_0$). Définir la sensibilité de déviation $S = y_{ec}/V_0$.
        \item Pour une tension alternative $V_d(t) = V_0 \sin(\omega t)$, on observe une trace lumineuse sur l'écran. En supposant que la persistance rétinienne et la rémanence du phosphore permettent de voir la courbe $y_{ec}(t)$, montrer que l'affichage n'est pas une sinusoïde parfaite $y_{ec} \propto \sin(\omega t)$ si $\omega$ n'est pas négligeable. Déterminer la condition sur $\omega$ (ou la période $T$) pour que l'approximation « quasi-statique » soit valide (erreur $< 5\%$ sur l'amplitude).
    \end{enumerate}
    \item \textbf{Optimisation du canon à électrons.} 
    Les électrons sont accélérés sous une tension d'anode $U_a$ ($v_0 = \sqrt{2eU_a/m_e}$). On souhaite maximiser la sensibilité $S$ tout en gardant une taille de tache acceptable (liée à $v_0$). 
    \begin{enumerate}
        \item Exprimer $S$ en fonction de $U_a$, $L$, $d$, $D$. 
        \item Si l'on double $U_a$, dans quel rapport $S$ change-t-elle ? Dans quel rapport la déflexion angulaire $\alpha$ (pour un $V_0$ fixe) change-t-elle ? Discuter du compromis entre sensibilité, bande passante (vitesse de balayage) et résolution spatiale.
    \end{enumerate}
\end{enumerate}
\end{exercice}

\begin{exercice}[Mouvement hélicoïdal et piège magnétique : synthèse]
Une particule de charge $q>0$ et de masse $m$ est injectée dans un champ magnétique uniforme $\vec{B} = B\,\vec{u}_z$ avec une vitesse initiale $\vec{v}_0 = v_\perp\,\vec{u}_x + v_\parallel\,\vec{u}_z$ ($v_\perp > 0$, $v_\parallel > 0$).

\begin{enumerate}
    \item \textbf{Caractéristiques du mouvement.} 
    \begin{enumerate}
        \item Écrire le vecteur vitesse $\vec{v}(t)$ et le vecteur position $\vec{r}(t)$ (origine au point d'injection). Identifier la nature de la trajectoire et donner son pas $p$ (avance axiale par tour).
        \item Démontrer que le moment cinétique par rapport à l'axe $(Oz)$, $L_z = m(x v_y - y v_x)$, est constant. L'interpréter physiquement (lien avec le moment magnétique $\mu$ de la spire de courant équivalente).
    \end{enumerate}
    \item \textbf{Miroir magnétique (piège à bouteille).} 
    Le champ magnétique n'est plus uniforme mais axisymétrique : $B_z(z) = B_0(1 + \alpha z^2)$ avec $\alpha > 0$, $B_r \approx -\frac{r}{2}\frac{dB_z}{dz}$ (approximation paraxiale, $\nabla \cdot \vec{B}=0$). La particule oscille entre deux points de retour $z = \pm z_m$.
    \begin{enumerate}
        \item En utilisant la conservation de l'énergie cinétique $E_c$ et du moment magnétique $\mu = \frac{m v_\perp^2}{2B}$ (invariant adiabatique), établir la condition de réflexion (point de retour) reliant $v_\parallel$, $v_\perp$, $B$ au point d'injection ($z=0$, $B=B_0$) et $B(z_m)$.
        \item En déduire l'expression de $z_m$ en fonction de $v_0$, $\theta_0$ (angle initial avec $\vec{B}$), $B_0$, $\alpha$.
        \item Déterminer l'angle critique $\theta_c$ (cône de perte) en dessous duquel la particule n'est pas piégée (sort du système par $z \to \infty$ où $B \to \infty$). Discuter de l'application pour le confinement de plasma (tokamak vs miroir magnétique).
    \end{enumerate}
\end{enumerate}
\end{exercice}
